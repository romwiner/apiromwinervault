async function register(event) {
  if (event) event.preventDefault();
  
  const emailInput = document.getElementById('email');
  const passInput = document.getElementById('pass');
  const termsCheck = document.getElementById('acceptTerms');
  
  const rawEmail = emailInput?.value?.trim();
  const pass = passInput?.value;
  const acceptTerms = termsCheck?.checked;
  
  // ✅ Sanitizar email antes de validar
  const email = sanitizeEmail(rawEmail);
  
  // Validaciones básicas
  if (!email || !pass) {
    log('⚠️ Ingresa correo y contraseña', 'warning');
    showNotification('⚠️ Completa todos los campos', 'warning');
    return;
  }
  
  // Validar formato de email
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    log('⚠️ Formato de email inválido', 'warning');
    showNotification('⚠️ Ingresa un email válido', 'warning');
    if (emailInput) {
      emailInput.classList.add('error');
      emailInput.setAttribute('aria-invalid', 'true');
    }
    return;
  }
  
  // ✅ Validar aceptación de términos con feedback accesible
  if (!acceptTerms) {
    log('⚠️ Debes aceptar Términos y Privacidad', 'warning');
    showNotification('⚠️ Acepta los Términos para continuar', 'warning');
    if (termsCheck) {
      termsCheck.classList.add('error');
      termsCheck.setAttribute('aria-invalid', 'true');
      setTimeout(() => {
        termsCheck.classList.remove('error');
        termsCheck.removeAttribute('aria-invalid');
      }, 2000);
    }
    return;
  }
  
  // ✅ Validación fuerte de contraseña con feedback específico
  const passErrors = validatePasswordStrength(pass);
  if (passErrors.length > 0) {
    const msg = '⚠️ Contraseña débil. Requiere: ' + passErrors.join(', ');
    log(msg, 'warning');
    showNotification(msg, 'warning');
    if (passInput) {
      passInput.classList.add('error');
      passInput.setAttribute('aria-invalid', 'true');
      passInput.focus();
      setTimeout(() => {
        passInput.classList.remove('error');
        passInput.removeAttribute('aria-invalid');
      }, 3000);
    }
    return;
  }
  
  // ✅ Referencia segura al botón
  const btn = event?.currentTarget || 
              document.querySelector('#authSection button[onclick*="register"]');
  
  try {
    // UI: estado de carga
    if (btn) {
      btn.disabled = true;
      btn.textContent = '✨ Creando...';
      btn.setAttribute('aria-busy', 'true');
    }
    if (emailInput) emailInput.classList.remove('error');
    if (passInput) passInput.classList.remove('error');
    
    // ✅ Registro: password nunca se loguea, email ya sanitizado
    await req('/register', 'POST', { email, password: pass });
    
    log('✅ Cuenta creada. Inicia sesión para comenzar.', 'success');
    
    // Limpiar campo de contraseña por seguridad
    if (passInput) passInput.value = '';
    
    showNotification('✅ Cuenta creada. Inicia sesión.', 'success');
    
  } catch (e) {
    // ✅ Mensajes de error específicos sin exponer detalles internos
    const msg = e.message.includes('409') ? '❌ Email ya registrado' : 
                e.message.includes('400') ? '❌ Datos inválidos' : 
                '❌ Error de registro';
    
    log(msg, 'error');
    showNotification(msg, 'error');
    
  } finally {
    // Restaurar UI siempre
    if (btn) {
      btn.disabled = false;
      btn.textContent = '✨ Registrar Cuenta Nueva';
      btn.removeAttribute('aria-busy');
    }
  }
}

function sanitizeForDisplay(str, maxLen = 100) {
  if (!str) return '';
  return String(str)
    .replace(/[<>]/g, '') // Remover tags HTML
    .replace(/javascript:/gi, '') // Prevenir protocolos peligrosos
    .substring(0, maxLen);
}

// ✅ Helper reutilizable para validar fortaleza de contraseña
function validatePasswordStrength(pass) {
  const errors = [];
  if (!pass || pass.length < 8) errors.push('mínimo 8 caracteres');
  if (!/[A-Z]/.test(pass)) errors.push('1 mayúscula');
  if (!/[0-9]/.test(pass)) errors.push('1 número');
  if (!/[^a-zA-Z0-9]/.test(pass)) errors.push('1 símbolo (!@#$%^&*)');
  return errors;
}

}

// 👤 Cargar perfil de usuario con validaciones de seguridad y sanitización
async function loadUser() {
  try {
    const p = await req('/api/profile');
    currentUser = p.user || p.profile;
    
    // Tier con fallback seguro y sanitización
    const rawTier = currentUser?.tier || localStorage.getItem('userTier') || 'personal';
    userTier = sanitizeForDisplay(rawTier, 20).replace(/[^a-z]/gi, '').toLowerCase() || 'personal';
    localStorage.setItem('userTier', userTier);
    
    // Admin: siempre verificar en backend, localStorage es solo UX
    isAdmin = p.user?.isAdmin === true;
    if (isAdmin) {
      localStorage.setItem('isAdmin', 'true');
    } else {
      localStorage.removeItem('isAdmin');
    }
    
    // ✅ Sanitizar nombre para mostrar (previene XSS)
    const rawName = p.profile?.displayName || p.user?.email?.split('@')?.[0] || 'Usuario';
    const safeName = sanitizeForDisplay(rawName, 50);
    
    // Actualizar UI con null checks y sanitización
    const emailEl = document.getElementById('uEmail');
    if (emailEl) {
      emailEl.textContent = safeName;
      emailEl.setAttribute('aria-label', `Usuario: ${safeName}`);
    }
    
    // ✅ Actualizar badge de tier con sanitización
    const tierBadge = document.getElementById('userTierBadge');
    if (tierBadge) {
      tierBadge.textContent = userTier.charAt(0).toUpperCase() + userTier.slice(1);
      tierBadge.className = `tier-badge tier-${userTier}`;
      tierBadge.setAttribute('aria-label', `Nivel de cuenta: ${userTier}`);
    }
    
    // ✅ Mostrar/ocultar funciones enterprise con ARIA
    const enterpriseFeatures = document.getElementById('enterpriseFeatures');
    if (enterpriseFeatures) {
      const isVisible = userTier === 'enterprise';
      enterpriseFeatures.classList.toggle('hidden', !isVisible);
      enterpriseFeatures.setAttribute('aria-hidden', isVisible ? 'false' : 'true');
    }
    
    // ✅ Panel Admin: solo si isAdmin real (verificado por backend)
    const adminPanel = document.getElementById('adminPanel');
    if (adminPanel) {
      adminPanel.style.display = isAdmin ? 'block' : 'none';
      adminPanel.setAttribute('aria-hidden', isAdmin ? 'false' : 'true');
    }
    
    // ✅ QR de referido: generar localmente (0% externos)
    const refQr = document.getElementById('refQr');
    if (refQr && typeof generateLocalQR === 'function') {
      const refCode = currentUser?.refCode || localStorage.getItem('refCode') || 'ROM000';
      const refLink = `${window.location.origin}?ref=${refCode}`;
      refQr.src = generateLocalQR(refLink, 160); // ✅ SVG embebido, sin HTTP externo
      refQr.alt = 'Código QR de referido';
      refQr.dataset.qr = refLink;
    }
    
    // ✅ Cuota de usuario con validación
    if (p.quota) {
      userQuota = p.quota;
      if (typeof updateQuotaDisplay === 'function') updateQuotaDisplay();
    }
    
  } catch (e) {
    // ✅ Manejo seguro de errores: no exponer detalles internos
    log('⚠️ Error cargando perfil: ' + sanitizeForDisplay(e.message, 100), 'warning');
    
    // Fallback: mostrar email básico si falla carga de perfil
    const email = localStorage.getItem('userEmail');
    const emailEl = document.getElementById('uEmail');
    if (emailEl && email) {
      const safeEmail = sanitizeForDisplay(email.split('@')[0], 30);
      emailEl.textContent = safeEmail;
    }
  }
}

  const fill = document.getElementById('quotaFill');

  }

  const limit = userQuota.tier?.storageLimitGB;

  if (limit === -1 || limit === Infinity) {

  }
  
  // ✅ Caso: límite definido
  if (limit && limit > 0) {
    const percent = Math.min(100, Math.max(0, (used / limit) * 100));
    const percentRounded = Math.round(percent);
    
    // Actualizar barra de progreso con atributos ARIA
    fill.style.width = percentRounded + '%';
    fill.setAttribute('aria-valuenow', percentRounded);
    fill.setAttribute('aria-valuemin', '0');
    fill.setAttribute('aria-valuemax', '100');
    
    // Actualizar texto con formato consistente
    text.textContent = `${used.toFixed(2)} GB / ${limit} GB`;
    
    // ✅ Alertas visuales y de accesibilidad según uso
    if (percent >= 90) {
      text.className = 'quota-critical';
      text.textContent += ' ⚠️ Casi lleno';
      text.setAttribute('aria-label', `Almacenamiento crítico: ${percentRounded}% usado`);
      // ✅ Notificación única para no saturar

        showNotification('⚠️ Tu almacenamiento está casi lleno (90%)', 'warning', 8000);
        display.dataset.criticalNotified = 'true';
      }
    } else if (percent >= 75) {
      text.className = 'quota-warning';
      text.textContent += ' ⚠️ Considera limpiar';
      text.setAttribute('aria-label', `Almacenamiento: ${percentRounded}% usado`);
      delete display.dataset.criticalNotified;
    } else {
      text.className = '';
      text.setAttribute('aria-label', `Almacenamiento: ${percentRounded}% usado`);
      delete display.dataset.criticalNotified;
    }
    
    display.classList.remove('hidden');
    display.setAttribute('aria-hidden', 'false');
    
  } else {
    // ✅ Caso: sin datos de quota
    display.classList.add('hidden');
    display.setAttribute('aria-hidden', 'true');
  }
}

// 🔐 Simular cifrado Envelope (local, sin conexiones externas)
async function simulateEnvelopeEncryption(titulo, categoria) {
  // ✅ En producción: usar WebCrypto API para cifrado real
  // Aquí simulamos con hash + timestamp para demo segura
  const data = `${titulo}|${categoria}|${Date.now()}|${Math.random().toString(36).slice(2)}`;
  
  // Hash simple (SHA-256 simulado para demo)
  let hash = 0;
  for (let i = 0; i < data.length; i++) {
    const char = data.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash = hash & hash;
  }
  
  // Retornar "contenido cifrado" simulado
  return `ENC:v1:${Math.abs(hash).toString(16)}:${btoa(data).slice(0, 50)}...`;
}

    return;

        progressFill.style.width = Math.round(prog) + '%';
        progressFill.setAttribute('aria-valuenow', Math.round(prog));
      }
    }, 250);
  };
  
  const stopProgress = () => {
    if (interval) {
      clearInterval(interval);
      interval = null;
    }
  };
  
  try {
    // ✅ Iniciar progreso simulado
    startProgress();
    
    // ✅ Preparar FormData con validación
    const formData = new FormData();
    formData.append('titulo', titulo);
    formData.append('categoria', cat);
    formData.append('forSale', forSale.toString());
    if (forSale) formData.append('price', price.toFixed(2));
    formData.append('archivo', file, file.name); // ✅ Nombre original preservado
    
    // ✅ Headers seguros

    // ✅ Fetch con AbortController para cancelación

      method: 'POST',
      headers,
      body: formData,
      signal: controller.signal,
      credentials: 'same-origin'

      progressFill.setAttribute('aria-valuenow', '100');

    }
    
    // ✅ Éxito
    if (status) {
      status.textContent = '✅ Archivo subido y cifrado';
      status.style.color = '#00e676';

      status.textContent = '❌ Error: ' + sanitizeForDisplay(errorMsg, 100);
      status.style.color = '#ff5252';
    }
    
    log('❌ Upload fallido: ' + sanitizeForDisplay(e.message, 100), 'error');
    showNotification(sanitizeForDisplay(errorMsg, 100), 'error');
    console.error('Upload error details:', { name: e.name, message: e.message, stack: e.stack });
    
  } finally {
    // ✅ Cleanup garantizado: siempre restaurar UI
    isUploading = false;
    stopProgress(); // ✅ Asegurar que el intervalo se limpie
    
    if (btn) {
      btn.disabled = false;
      btn.textContent = '✅ Subir y Cifrar';
      btn.removeAttribute('aria-busy');
    }
  }

// ✅ Helper para sanitizar texto antes de insertar en DOM/logs (previene XSS)
function sanitizeForDisplay(str, maxLen = 100) {
  if (!str) return '';
  return String(str)
    .replace(/[&<>"']/g, (m) => ({'&':'&','<':'<','>':'>','"':'"',"'":'''}[m]))
    .substring(0, maxLen);
}

// ✅ Helper para sanitizar IDs usados en onclick (previene inyección de código)
function sanitizeIdForEvent(str) {
  if (!str) return '';
  return String(str).replace(/[^a-zA-Z0-9\-_]/g, '').substring(0, 50);
}

      textarea.select();
      document.execCommand('copy');
      document.body.removeChild(textarea);
      showNotification('✅ Enlace copiado', 'success');
    }
    
    log(`🔗 Share generado para: ${safeId}`, 'info');
    
  } catch (e) {
    // ✅ Mensaje seguro: no exponer detalles internos
    const safeMsg = 'Error al generar enlace de compartir';

// ✅ Helper para validar URLs seguras (solo same-origin o rutas relativas)
function isValidInternalUrl(url) {
  if (!url) return false;
  try {
    // Permitir rutas relativas
    if (url.startsWith('/')) return true;
    // Permitir same-origin
    const parsed = new URL(url, window.location.origin);
    return parsed.origin === window.location.origin;
  } catch {
    return false;
  }
}

   // ✅ Helper para sanitizar texto antes de insertar en DOM/logs (previene XSS)
function sanitizeForDisplay(str, maxLen = 100) {
  if (!str) return '';
  return String(str)
    .replace(/[&<>"']/g, (m) => ({'&':'&','<':'<','>':'>','"':'"',"'":'''}[m]))
    .substring(0, maxLen);
}

// ✅ Helper para sanitizar IDs usados en eventos (previene inyección de código)
function sanitizeIdForEvent(str) {
  if (!str) return '';
  return String(str).replace(/[^a-zA-Z0-9\-_]/g, '').substring(0, 50);
}

// ✅ Helper para validar URLs de imágenes (solo permite same-origin o data URLs)
function isValidImageUrl(url) {
  if (!url) return false;
  // Permitir data URLs seguras
  if (url.startsWith('data:image/')) return true;
  // Permitir rutas relativas
  if (url.startsWith('/')) return true;
  // Permitir same-origin
  try {
    const parsed = new URL(url, window.location.origin);
    return parsed.origin === window.location.origin;
  } catch {
    return false;
  }
}

// ✅ FUNCIÓN req() CORREGIDA - Acepta ambos formatos de llamada
async function req(url, metodoOpciones, bodyOpcional) {
  try {
    let opciones = {};
    
    // Soporta ambos formatos:
    // Formato 1: req(url, { method: 'POST', body: {...} })
    // Formato 2: req(url, 'POST', { message: '...' })
    if (typeof metodoOpciones === 'string') {
      opciones = {
        method: metodoOpciones,
        body: bodyOpcional ? JSON.stringify(bodyOpcional) : undefined
      };
    } else if (typeof metodoOpciones === 'object') {
      opciones = {
        method: metodoOpciones.method || 'GET',
        body: metodoOpciones.body ? JSON.stringify(metodoOpciones.body) : undefined
      };
    }

    const respuesta = await fetch(url, {
      method: opciones.method || 'GET',
      headers: { 
        'Content-Type': 'application/json',
        ...(opciones.headers || {})
      },
      body: opciones.body
    });

    if (!respuesta.ok) {
      const textoError = await respuesta.text().catch(() => '');
      throw new Error(`HTTP ${respuesta.status}: ${respuesta.statusText} ${textoError}`);
    }

    const texto = await respuesta.text();
    return texto ? JSON.parse(texto) : {};
  } catch (error) {
    log(`❌ Error en ${url}: ${error.message}`, 'error');
    throw error;
  }
}

// ✅ NUEVA FUNCIÓN: Sanitizar texto para mostrar en pantalla (evita XSS)
function sanitizeForDisplay(texto, maxLongitud = 100) {
  if (texto === null || texto === undefined) return '';
  let limpio = String(texto);
  // Escapar caracteres HTML peligrosos
  limpio = limpio
    .replace(/&/g, '&')
    .replace(//g, '>')
    .replace(/"/g, '"')
    .replace(/'/g, ''');
  // Limitar longitud
  if (maxLongitud && limpio.length > maxLongitud) {
    limpio = limpio.substring(0, maxLongitud) + '...';
  }
  return limpio;
}

// Inicializar canvas cuando el DOM esté listo
function initSignatureCanvas() {
  if (!sigCanvas) {
    log('⚠️ Canvas de firma no encontrado', 'warning');
    return;
  }

  sigContext = sigCanvas.getContext('2d');
  
  // ✅ Configurar estilo del trazo
  sigContext.strokeStyle = '#ffffff';
  sigContext.lineWidth = 2.5;
  sigContext.lineCap = 'round';
  sigContext.lineJoin = 'round';

  // ✅ Función para obtener posición del cursor/tacto
  function getPosition(evento) {
    const rect = sigCanvas.getBoundingClientRect();
    const clienteX = evento.touches?.[0]?.clientX || evento.clientX;
    const clienteY = evento.touches?.[0]?.clientY || evento.clientY;
    return {
      x: clienteX - rect.left,
      y: clienteY - rect.top
    };
  }

  // ✅ Usar addEventListener (no sobrescribe otros handlers)
  sigCanvas.addEventListener('mousedown', (e) => {
    isDrawing = true;
    const pos = getPosition(e);
    sigContext.beginPath();
    sigContext.moveTo(pos.x, pos.y);
  });

  sigCanvas.addEventListener('mousemove', (e) => {
    if (!isDrawing) return;
    const pos = getPosition(e);
    sigContext.lineTo(pos.x, pos.y);
    sigContext.stroke();
  });

  sigCanvas.addEventListener('mouseup', () => {
    isDrawing = false;
  });

  sigCanvas.addEventListener('mouseleave', () => {
    isDrawing = false;
  });

  // Soporte táctil
  sigCanvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    isDrawing = true;
    const pos = getPosition(e);
    sigContext.beginPath();
    sigContext.moveTo(pos.x, pos.y);
  }, { passive: false });

  sigCanvas.addEventListener('touchmove', (e) => {
    e.preventDefault();
    if (!isDrawing) return;
    const pos = getPosition(e);
    sigContext.lineTo(pos.x, pos.y);
    sigContext.stroke();
  }, { passive: false });

  sigCanvas.addEventListener('touchend', () => {
    isDrawing = false;
  });

  log('✅ Canvas de firma inicializado', 'success');
}

async function handleAvatar(evento) {
  const archivo = evento.target.files[0];
  if (!archivo) return;

  // ✅ Validar tipo de archivo
  const tiposPermitidos = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
  if (!tiposPermitidos.includes(archivo.type)) {
    showNotification('❌ Solo se permiten imágenes (JPG, PNG, GIF, WebP)', 'error');
    evento.target.value = ''; // Limpiar input
    return;
  }

  // ✅ Validar tamaño (máximo 5MB)
  const tamanoMaximo = 5 * 1024 * 1024; // 5MB
  if (archivo.size > tamanoMaximo) {
    showNotification('❌ La imagen es muy grande (máx. 5MB)', 'error');
    evento.target.value = '';
    return;
  }

  const avatarImg = document.getElementById('userAvatar');
  if (!avatarImg) {
    showNotification('❌ Elemento de avatar no encontrado', 'error');
    return;
  }

  // Leer y guardar la imagen
  const reader = new FileReader();
  reader.onload = (e) => {
    avatarImg.src = e.target.result;
    try {
      localStorage.setItem('romwiner_avatar', e.target.result);
      showNotification('✅ Foto guardada en tu dispositivo', 'success');
    } catch (error) {
      // localStorage puede estar lleno
      log('⚠️ No se pudo guardar en localStorage: ' + error.message, 'warning');
      showNotification('✅ Foto cargada (no se pudo guardar permanentemente)', 'warning');
    }
  };
  
  reader.onerror = () => {
    showNotification('❌ Error al leer la imagen', 'error');
  };
  
  reader.readAsDataURL(archivo);
}

// ============================================
// 📱 MÓDULO QR COMPLETO CON REED-SOLOMON
// Implementación basada en ISO/IEC 18004
// ============================================

const QRModule = (function() {
  'use strict';

  // ============================================
  // 🧮 CAMPO DE GALOIS GF(256)
  // ============================================
  class GaloisField {
    constructor() {
      // Tablas de exponenciales y logaritmos para GF(256)
      this.exp = new Array(512);
      this.log = new Array(256);
      
      // Generar tablas usando el polinomio primitivo 0x11d
      let x = 1;
      for (let i = 0; i < 255; i++) {
        this.exp[i] = x;
        this.log[x] = i;
        x = x << 1;
        if (x & 0x100) {
          x = x ^ 0x11d; // Polinomio primitivo
        }
      }
      
      // Extender tabla de exponenciales
      for (let i = 255; i < 512; i++) {
        this.exp[i] = this.exp[i - 255];
      }
    }

    // Multiplicación en GF(256)
    multiply(x, y) {
      if (x === 0 || y === 0) return 0;
      return this.exp[this.log[x] + this.log[y]];
    }

    // División en GF(256)
    divide(x, y) {
      if (y === 0) throw new Error('División por cero');
      if (x === 0) return 0;
      return this.exp[(this.log[x] + 255 - this.log[y]) % 255];
    }

    // Potencia en GF(256)
    power(x, power) {
      if (x === 0) return 0;
      return this.exp[(this.log[x] * power) % 255];
    }

    // Inverso multiplicativo
    inverse(x) {
      if (x === 0) throw new Error('Inverso de cero');
      return this.exp[255 - this.log[x]];
    }
  }

  const GF = new GaloisField();

  // ============================================
  // 🧮 POLINOMIO REED-SOLOMON
  // ============================================
  class ReedSolomonEncoder {
    constructor(numECCodeWords) {
      this.numECCodeWords = numECCodeWords;
      this.generator = this.computeGenerator();
    }

    // Calcular polinomio generador
    computeGenerator() {
      let gen = [1];
      for (let i = 0; i < this.numECCodeWords; i++) {
        const newGen = new Array(gen.length + 1).fill(0);
        const factor = GF.exp[i];
        
        for (let j = 0; j < gen.length; j++) {
          newGen[j] ^= gen[j];
          newGen[j + 1] ^= GF.multiply(gen[j], factor);
        }
        
        gen = newGen;
      }
      
      return gen;
    }

    // Codificar datos con Reed-Solomon
    encode(data) {
      const result = new Array(this.numECCodeWords).fill(0);
      
      for (let i = 0; i < data.length; i++) {
        const factor = data[i] ^ result[0];
        result.shift();
        result.push(0);
        
        for (let j = 0; j < result.length; j++) {
          result[j] ^= GF.multiply(this.generator[j + 1], factor);
        }
      }
      
      return result;
    }
  }

  // ============================================
  // 📊 TABLAS DE CAPACIDAD Y CORRECCIÓN
  // ============================================
  
  // Capacidades por versión y nivel de corrección (modo byte)
  const CAPACITIES = {
    1: { L: 17, M: 14, Q: 11, H: 7 },
    2: { L: 32, M: 26, Q: 20, H: 14 },
    3: { L: 53, M: 42, Q: 32, H: 24 },
    4: { L: 78, M: 62, Q: 46, H: 34 },
    5: { L: 106, M: 84, Q: 60, H: 44 },
    6: { L: 134, M: 106, Q: 74, H: 58 },
    7: { L: 154, M: 122, Q: 86, H: 64 },
    8: { L: 192, M: 152, Q: 108, H: 84 },
    9: { L: 230, M: 180, Q: 130, H: 98 },
    10: { L: 271, M: 213, Q: 151, H: 119 }
  };

  // Palabras de código de corrección por versión y nivel
  const EC_CODEWORDS = {
    1: { L: 7, M: 10, Q: 13, H: 17 },
    2: { L: 10, M: 16, Q: 22, H: 28 },
    3: { L: 15, M: 26, Q: 36, H: 44 },
    4: { L: 20, M: 36, Q: 52, H: 64 },
    5: { L: 26, M: 48, Q: 72, H: 88 },
    6: { L: 36, M: 64, Q: 96, H: 112 },
    7: { L: 40, M: 72, Q: 108, H: 130 },
    8: { L: 48, M: 88, Q: 132, H: 156 },
    9: { L: 60, M: 110, Q: 160, H: 192 },
    10: { L: 72, M: 130, Q: 192, H: 224 }
  };

  // ============================================
  // 🔧 FUNCIONES AUXILIARES
  // ============================================

  // Determinar versión óptima
  function getVersion(dataLength, ecLevel = 'M') {
    for (let v = 1; v <= 10; v++) {
      if (CAPACITIES[v][ecLevel] >= dataLength) {
        return v;
      }
    }
    return -1; // Datos muy grandes
  }

  // Codificar datos en modo byte
  function encodeData(text, version, ecLevel) {
    const dataBits = [];
    
    // Indicador de modo byte (0100)
    dataBits.push(0, 1, 0, 0);
    
    // Longitud (8 bits para versión 1-9, 16 bits para 10+)
    const len = text.length;
    const lenBits = version <= 9 ? 8 : 16;
    for (let i = lenBits - 1; i >= 0; i--) {
      dataBits.push((len >> i) & 1);
    }
    
    // Datos (UTF-8)
    const encoder = new TextEncoder();
    const bytes = encoder.encode(text);
    for (const byte of bytes) {
      for (let j = 7; j >= 0; j--) {
        dataBits.push((byte >> j) & 1);
      }
    }
    
    // Terminador
    const totalDataBits = getTotalDataBits(version, ecLevel);
    const terminatorLen = Math.min(4, totalDataBits - dataBits.length);
    for (let i = 0; i < terminatorLen; i++) {
      dataBits.push(0);
    }
    
    // Rellenar hasta múltiplo de 8
    while (dataBits.length % 8 !== 0) {
      dataBits.push(0);
    }
    
    // Convertir a bytes
    const dataBytes = [];
    for (let i = 0; i < dataBits.length; i += 8) {
      let byte = 0;
      for (let j = 0; j < 8; j++) {
        byte = (byte << 1) | (dataBits[i + j] || 0);
      }
      dataBytes.push(byte);
    }
    
    // Rellenar con patrones de padding
    const totalBytes = Math.floor(totalDataBits / 8);
    const padBytes = [0xEC, 0x11];
    let padIdx = 0;
    while (dataBytes.length < totalBytes) {
      dataBytes.push(padBytes[padIdx % 2]);
      padIdx++;
    }
    
    return dataBytes;
  }

  // Calcular bits de datos totales
  function getTotalDataBits(version, ecLevel) {
    const size = version * 4 + 17;
    let totalModules = size * size;
    
    // Restar módulos de función
    totalModules -= 8 * 8 * 3; // Finders
    totalModules -= 15 * 2 + 9; // Timing
    totalModules -= 31; // Format info
    if (version >= 2) totalModules -= 25; // Alignment
    
    const totalCodewords = Math.floor(totalModules / 8);
    const ecCodewords = EC_CODEWORDS[version][ecLevel];
    
    return (totalCodewords - ecCodewords) * 8;
  }

  // ============================================
  // 🎨 GENERACIÓN DE MATRIZ QR
  // ============================================

  function generateMatrix(text, ecLevel = 'M') {
    const version = getVersion(text.length, ecLevel);
    if (version === -1) return null;

    const size = version * 4 + 17;
    const matrix = Array.from({ length: size }, () => Array(size).fill(null));
    const reserved = Array.from({ length: size }, () => Array(size).fill(false));

    // Dibujar patrones de finder (esquinas)
    function drawFinder(row, col) {
      for (let r = -1; r <= 7; r++) {
        for (let c = -1; c <= 7; c++) {
          const rr = row + r, cc = col + c;
          if (rr < 0 || rr >= size || cc < 0 || cc >= size) continue;
          
          if (r === -1 || r === 7 || c === -1 || c === 7) {
            matrix[rr][cc] = 0; // Separador blanco
          } else if (
            (r === 0 || r === 6) || (c === 0 || c === 6) ||
            (r >= 2 && r <= 4 && c >= 2 && c <= 4)
          ) {
            matrix[rr][cc] = 1; // Patrón negro
          } else {
            matrix[rr][cc] = 0;
          }
          reserved[rr][cc] = true;
        }
      }
    }

    // Tres finders
    drawFinder(0, 0);
    drawFinder(0, size - 7);
    drawFinder(size - 7, 0);

    // Patrones de sincronización
    for (let i = 8; i < size - 8; i++) {
      const val = i % 2 === 0 ? 1 : 0;
      if (!reserved[6][i]) {
        matrix[6][i] = val;
        reserved[6][i] = true;
      }
      if (!reserved[i][6]) {
        matrix[i][6] = val;
        reserved[i][6] = true;
      }
    }

    // Módulo oscuro
    matrix[size - 8][8] = 1;
    reserved[size - 8][8] = true;

    // Reservar áreas de formato
    for (let i = 0; i < 9; i++) {
      if (i < size) {
        reserved[8][i] = true;
        reserved[i][8] = true;
      }
    }
    for (let i = 0; i < 8; i++) {
      reserved[8][size - 1 - i] = true;
      reserved[size - 1 - i][8] = true;
    }

    // Codificar datos
    const dataBytes = encodeData(text, version, ecLevel);
    
    // Aplicar Reed-Solomon
    const ecCodewords = EC_CODEWORDS[version][ecLevel];
    const rsEncoder = new ReedSolomonEncoder(ecCodewords);
    const ecBytes = rsEncoder.encode(dataBytes);
    
    // Combinar datos + corrección
    const allBytes = [...dataBytes, ...ecBytes];
    
    // Convertir a bits
    const allBits = [];
    for (const byte of allBytes) {
      for (let i = 7; i >= 0; i--) {
        allBits.push((byte >> i) & 1);
      }
    }

    // Colocar datos en zigzag
    let bitIndex = 0;
    let col = size - 1;
    let goingUp = true;

    while (col > 0) {
      if (col === 6) col--;
      
      const rowRange = goingUp 
        ? Array.from({ length: size }, (_, i) => size - 1 - i)
        : Array.from({ length: size }, (_, i) => i);
      
      for (const row of rowRange) {
        for (let i = 0; i < 2; i++) {
          const c = col - i;
          if (c < 0 || reserved[row][c]) continue;
          
          matrix[row][c] = bitIndex < allBits.length ? allBits[bitIndex++] : 0;
        }
      }
      
      goingUp = !goingUp;
      col -= 2;
    }

    // Aplicar máscara 0
    for (let r = 0; r < size; r++) {
      for (let c = 0; c < size; c++) {
        if (!reserved[r][c] && (r + c) % 2 === 0) {
          matrix[r][c] ^= 1;
        }
      }
    }

    // Colocar bits de formato (nivel de corrección + máscara)
    const formatBits = getFormatBits(ecLevel, 0);
    placeFormatBits(matrix, size, formatBits);

    return matrix;
  }

  // Obtener bits de formato
  function getFormatBits(ecLevel, maskPattern) {
    const ecLevelBits = { L: 1, M: 0, Q: 3, H: 2 }[ecLevel];
    const data = (ecLevelBits << 3) | maskPattern;
    
    // Calcular BCH(15,5)
    let format = data << 10;
    const generator = 0x537; // Polinomio generador BCH
    
    for (let i = 4; i >= 0; i--) {
      if (format & (1 << (i + 10))) {
        format ^= generator << i;
      }
    }
    
    format = ((data << 10) | format) ^ 0x5412; // Máscara XOR
    
    // Convertir a array de bits
    const bits = [];
    for (let i = 14; i >= 0; i--) {
      bits.push((format >> i) & 1);
    }
    
    return bits;
  }

  // Colocar bits de formato en la matriz
  function placeFormatBits(matrix, size, formatBits) {
    // Alrededor del finder superior-izquierdo
    const positions1 = [
      [8, 0], [8, 1], [8, 2], [8, 3], [8, 4], [8, 5], [8, 7], [8, 8],
      [7, 8], [5, 8], [4, 8], [3, 8], [2, 8], [1, 8], [0, 8]
    ];
    
    // Alrededor de los otros finders
    const positions2 = [
      [size - 1, 8], [size - 2, 8], [size - 3, 8], [size - 4, 8],
      [size - 5, 8], [size - 6, 8], [size - 7, 8],
      [8, size - 8], [8, size - 7], [8, size - 6], [8, size - 5],
      [8, size - 4], [8, size - 3], [8, size - 2], [8, size - 1]
    ];
    
    for (let i = 0; i < 15; i++) {
      if (i < positions1.length) {
        const [r, c] = positions1[i];
        if (r < size && c < size) matrix[r][c] = formatBits[i];
      }
      if (i < positions2.length) {
        const [r, c] = positions2[i];
        if (r < size && c < size) matrix[r][c] = formatBits[i];
      }
    }
  }

  // ============================================
  // 🎨 FUNCIÓN PÚBLICA: GENERAR QR
  // ============================================

  function generate(text, size = 200, moduleColor = '#000000', bgColor = '#ffffff', ecLevel = 'M') {
    // Validar entrada
    if (!text || text.length === 0) {
      console.warn('⚠️ QRModule: texto vacío');
      return null;
    }

    if (text.length > 200) {
      console.warn('⚠️ QRModule: texto muy largo, truncando a 200 caracteres');
      text = text.substring(0, 200);
    }

    // Validar nivel de corrección
    if (!['L', 'M', 'Q', 'H'].includes(ecLevel)) {
      console.warn('⚠️ QRModule: nivel de corrección inválido, usando M');
      ecLevel = 'M';
    }

    try {
      // Generar matriz con Reed-Solomon
      const matrix = generateMatrix(text, ecLevel);
      if (!matrix) {
        console.warn('⚠️ QRModule: no se pudo generar la matriz');
        return null;
      }

      const qrSize = matrix.length;
      const canvas = document.createElement('canvas');
      const padding = 4;
      const totalSize = qrSize + padding * 2;
      const moduleSize = size / totalSize;

      canvas.width = canvas.height = size;
      const ctx = canvas.getContext('2d');

      // Fondo
      ctx.fillStyle = bgColor;
      ctx.fillRect(0, 0, size, size);

      // Módulos
      ctx.fillStyle = moduleColor;
      for (let r = 0; r < qrSize; r++) {
        for (let c = 0; c < qrSize; c++) {
          if (matrix[r][c] === 1) {
            ctx.fillRect(
              Math.round((c + padding) * moduleSize),
              Math.round((r + padding) * moduleSize),
              Math.ceil(moduleSize),
              Math.ceil(moduleSize)
            );
          }
        }
      }

      console.log(`✅ QR generado: ${text.length} chars, versión ${getVersion(text.length, ecLevel)}, EC: ${ecLevel}`);
      return canvas.toDataURL('image/png');

    } catch (error) {
      console.error('❌ QRModule error:', error.message);
      return null;
    }
  }

  // API pública
  return Object.freeze({
    generate: generate,
    version: '2.0.0-reed-solomon',
    note: 'QR completo con Reed-Solomon - Escaneable por lectores reales'
  });

})();

  const elemento = document.getElementById('idQR');
  if (!elemento) {
    log('⚠️ Elemento #idQR no encontrado', 'warning');
    return;
  }

  // ✅ Obtener datos del usuario de forma segura con validaciones
  const usuario = typeof currentUser !== 'undefined' ? currentUser : {};
  const emailElemento = document.getElementById('uEmail');
  const avatarElemento = document.getElementById('userAvatar');
  const multisigElemento = document.getElementById('multisigToggle');
  const filasContactos = document.querySelectorAll('.contact-row');

  // ✅ Sanitizar datos antes de usarlos
  const datosIdentidad = {
    id: sanitizeForDisplay(usuario.uid || 'ROM' + Date.now(), 50),
    nombre: sanitizeForDisplay(emailElemento?.textContent?.trim() || 'Usuario', 50),
    avatar: avatarElemento?.src || '',
    sigHash: typeof sigData !== 'undefined' && sigData 
      ? await hashLocal(sigData, true) 
      : 'none',
    contactosRecuperacion: Number(filasContactos.length) || 0,
    multisig: Boolean(multisigElemento?.checked),
    timestamp: Date.now()
  };

  if (tipo === 'qr') {
    // ✅ QR REAL escaneable
    try {
      // ✅ Verificar que QRModule exista antes de usarlo
      if (typeof QRModule === 'undefined' || typeof QRModule.generate !== 'function') {
        throw new Error('QRModule no disponible');
      }

      // ✅ Codificar datos como string compacto con sanitización
      const datosQR = JSON.stringify({
        id: datosIdentidad.id,
        n: datosIdentidad.nombre.substring(0, 20),
        h: datosIdentidad.sigHash.substring(0, 16),
        t: datosIdentidad.timestamp
      });

      // ✅ Validar longitud de datos (QR tiene límite)
      if (datosQR.length > 200) {
        throw new Error('Datos demasiado grandes para QR');
      }

      const qrImagen = QRModule.generate(
        datosQR,
        200,
        '#00e676',  // Color del módulo (verde)
        '#0b0c10'   // Color de fondo (oscuro)
      );

      if (qrImagen) {
        elemento.src = qrImagen;
        elemento.alt = 'QR de Identidad (escaneable)';
        elemento.style.borderRadius = '8px';
        log('📱 QR real generado y escaneable', 'success');
        showNotification('✅ QR generado correctamente', 'success');
      } else {
        throw new Error('QRModule retornó imagen vacía');
      }
    } catch (error) {
      log('⚠️ QR falló, usando identicon: ' + sanitizeForDisplay(error.message, 100), 'warning');
      
      // ✅ Verificar que generateIdenticon exista antes de usarlo
      if (typeof generateIdenticon === 'function') {
        await generateIdenticon('idQR', datosIdentidad);
      } else {
        log('❌ generateIdenticon no disponible', 'error');
        showNotification('❌ No se pudo generar código de identidad', 'error');
      }
    }
  } else {
    // ✅ Identicon (patrón visual único)
    if (typeof generateIdenticon === 'function') {
      await generateIdenticon('idQR', datosIdentidad);
    } else {
      log('❌ generateIdenticon no disponible', 'error');
      showNotification('❌ No se pudo generar identicon', 'error');
    }
  }
}

async function toggleMultisig() {

    return;

    showNotification('⚠️ Debes iniciar sesión', 'warning');
    checkbox.checked = !checkbox.checked; // Revertir cambio visual
    return;
  }

  const activado = checkbox.checked;
  
  try {
    showNotification(activado ? '🔄 Activando multifirma...' : '🔄 Desactivando multifirma...', 'info');

    if (res?.status === 401) {
      checkbox.checked = !activado; // Revertir checkbox
      if (typeof handleSessionExpired === 'function') {
        handleSessionExpired();
      } else {
        showNotification('⚠️ Sesión expirada. Inicia sesión nuevamente.', 'warning');
      }
      return;
    }
    
    showNotification(
      activado ? '✅ Multifirma activada' : '🔓 Multifirma desactivada', 
      'success'
    );
    log(activado ? '🔐 Multifirma ACTIVADA' : '🔓 Multifirma DESACTIVADA', 'success');
  } catch (error) {
    // ✅ Revertir el checkbox si falla
    checkbox.checked = !activado;

    const safeMsg = sanitizeForDisplay(error.message, 100);
    showNotification('❌ Error: ' + safeMsg, 'error');
    log('❌ Error en multifirma: ' + safeMsg, 'error');

function showProfileSection() {

    return;
  }

  // ✅ Token consistente (verifica múltiples claves)
  const token = typeof window.token !== 'undefined' ? window.token 
              : localStorage.getItem('token') 
              || localStorage.getItem('auth_token');
  
  if (!token) {
    showNotification('⚠️ Debes iniciar sesión primero', 'warning');
    return;
  }

  panel.classList.remove('hidden');
  panel.setAttribute('aria-hidden', 'false'); // ✅ Accesibilidad

  if (typeof loadContacts === 'function') {
    loadContacts();
  }
  
  // Generar QR/Identicon si la función existe
  if (typeof genQR === 'function') {
    genQR();
  }
  
  // Cargar perfil del usuario
  req('/api/profile')
    .then(respuesta => {
      if (!respuesta || typeof respuesta !== 'object') {
        log('⚠️ Respuesta de perfil inválida', 'warning');
        return;
      }

      const user = respuesta.user || respuesta.profile;
      if (!user) {
        log('⚠️ Sin datos de usuario en respuesta', 'warning');
        return;
      }

      const avatarElemento = document.getElementById('userAvatar');
      const emailElemento = document.getElementById('uEmail');
      
      // ✅ VALIDACIÓN ESTRICTA DE AVATAR (solo data URLs o same-origin)
      if (user.avatar && avatarElemento) {
        const avatarUrl = String(user.avatar);
        const esDataUrl = /^data:image\/(png|jpeg|jpg|gif|webp);base64,/.test(avatarUrl);
        const esMismoOrigen = avatarUrl.startsWith('/') || 
                             avatarUrl.startsWith(window.location.origin);
        
        if (esDataUrl || esMismoOrigen) {
          avatarElemento.src = avatarUrl;
          avatarElemento.setAttribute('loading', 'lazy'); // ✅ Rendimiento
          avatarElemento.setAttribute('aria-label', 'Avatar de usuario');
        } else {
          // ✅ Rechazar URL externa: usar placeholder seguro
          avatarElemento.src = 'data:image/svg+xml,%3Csvg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"%3E%3Ccircle cx="50" cy="50" r="40" fill="%23151922" stroke="%2300e676" stroke-width="4"/%3E%3Ctext x="50" y="65" font-size="30" fill="%2300e676" text-anchor="middle"%3E👤%3C/text%3E%3C/svg%3E';
          avatarElemento.setAttribute('aria-label', 'Avatar predeterminado');
          log('⚠️ Avatar externo bloqueado por seguridad (cero enlaces externos)', 'warning');
        }
      }
      
      // ✅ Sanitizar email antes de mostrar
      if (user.email && emailElemento) {
        const safeEmail = typeof sanitizeForDisplay === 'function' 
          ? sanitizeForDisplay(user.email, 100)
          : String(user.email).replace(/[<>]/g, '').substring(0, 100);
        emailElemento.textContent = safeEmail;
        emailElemento.setAttribute('aria-label', `Usuario: ${safeEmail}`);
      }
      
      log('✅ Perfil cargado correctamente', 'success');
    })
    .catch(error => {
      // ✅ Manejo específico según tipo de error
      const msg = error.message || '';
      if (msg.includes('401')) {
        log('⚠️ Sesión expirada al cargar perfil', 'warning');
        showNotification('⚠️ Sesión expirada. Inicia sesión nuevamente.', 'warning');
        if (typeof handleSessionExpired === 'function') handleSessionExpired();
      } else if (msg.includes('403')) {
        log('⚠️ Acceso denegado al perfil', 'warning');
      } else {
        const safeMsg = typeof sanitizeForDisplay === 'function'
          ? sanitizeForDisplay(msg, 100)
          : String(msg).substring(0, 100);
        log('⚠️ Error cargando perfil: ' + safeMsg, 'warning');
      }
    });
}

// ✅ Helper: Sanitizar label para canvas
function sanitizeLabel(label, maxLen = 20) {
  if (!label) return '';
  return String(label)
    .replace(/[<>"']/g, '')
    .replace(/javascript:/gi, '')
    .substring(0, maxLen);
}

// ✅ Helper: Validar y limpiar datos numéricos
function cleanNumericData(data) {
  if (!Array.isArray(data)) return [];
  return data.map(d => {
    const n = Number(d);
    return isNaN(n) ? 0 : n;
  });
}

// Función para dibujar gráfico de líneas nativo (CORREGIDA)
function drawLineChart(canvasId, labels, data, color = '#00e676') {
  const canvas = document.getElementById(canvasId);
  if (!canvas) {
    console.warn('⚠️ Canvas no encontrado:', canvasId);
    return null;
  }

  // ✅ Validación robusta de inputs
  if (!Array.isArray(labels)) labels = [];
  const datosLimpios = cleanNumericData(data);
  
  // ✅ Asegurar misma longitud
  const maxLength = Math.max(labels.length, datosLimpios.length);
  const labelsFinales = Array.from({ length: maxLength }, (_, i) => 
    sanitizeLabel(labels[i] || `Punto ${i + 1}`, 15)
  );
  const datosFinales = Array.from({ length: maxLength }, (_, i) => 
    datosLimpios[i] || 0
  );

  const ctx = canvas.getContext('2d');
  
  // ✅ Soporte para pantallas retina (devicePixelRatio)
  const dpr = window.devicePixelRatio || 1;
  const displayWidth = canvas.offsetWidth || canvas.width || 600;
  const displayHeight = canvas.height || 300;
  
  canvas.width = displayWidth * dpr;
  canvas.height = displayHeight * dpr;
  canvas.style.width = displayWidth + 'px';
  canvas.style.height = displayHeight + 'px';
  ctx.scale(dpr, dpr);
  
  const width = displayWidth;
  const height = displayHeight;
  const padding = 40;

  // ✅ Accesibilidad ARIA
  canvas.setAttribute('role', 'img');

  // Limpiar canvas
  ctx.clearRect(0, 0, width, height);

  // ✅ Mensaje si no hay datos
  if (datosFinales.length === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('📭 Sin datos disponibles', width / 2, height / 2);
    canvas.setAttribute('aria-label', 'Gráfico sin datos');
    return null;
  }

  // ✅ Calcular escalas con protección contra NaN
  const maxValue = Math.max(...datosFinales);
  const minValue = Math.min(0, ...datosFinales);
  const range = maxValue - minValue || 1;

  // ✅ Actualizar aria-label con datos reales
  canvas.setAttribute('aria-label', 
    `Gráfico de líneas: ${datosFinales.length} puntos, máximo $${maxValue.toFixed(2)}, mínimo $${minValue.toFixed(2)}`
  );

  // Dibujar ejes
  ctx.strokeStyle = 'rgba(255,255,255,0.1)';
  ctx.lineWidth = 1;
  
  // Eje Y
  ctx.beginPath();
  ctx.moveTo(padding, padding);
  ctx.lineTo(padding, height - padding);
  ctx.stroke();
  
  // Eje X
  ctx.beginPath();
  ctx.moveTo(padding, height - padding);
  ctx.lineTo(width - padding, height - padding);
  ctx.stroke();

  // Dibujar líneas de grid horizontales
  ctx.strokeStyle = 'rgba(255,255,255,0.05)';
  for (let i = 0; i <= 4; i++) {
    const y = padding + (height - 2 * padding) * (i / 4);
    ctx.beginPath();
    ctx.moveTo(padding, y);
    ctx.lineTo(width - padding, y);
    ctx.stroke();
    
    // Etiquetas Y
    const valor = maxValue - (range * i / 4);
    ctx.fillStyle = '#8b95a5';
    ctx.font = '10px system-ui';
    ctx.textAlign = 'right';
    ctx.fillText('$' + valor.toFixed(0), padding - 5, y + 3);
  }

  // ✅ Protección contra división por cero
  const stepX = datosFinales.length > 1 
    ? (width - 2 * padding) / (datosFinales.length - 1) 
    : 0;
  
  // Área bajo la línea con gradiente
  const gradient = ctx.createLinearGradient(0, padding, 0, height - padding);
  gradient.addColorStop(0, color + '40');
  gradient.addColorStop(1, color + '05');
  ctx.fillStyle = gradient;
  ctx.beginPath();
  ctx.moveTo(padding, height - padding);
  
  datosFinales.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    ctx.lineTo(x, y);
  });
  
  ctx.lineTo(padding + (datosFinales.length - 1) * stepX, height - padding);
  ctx.closePath();
  ctx.fill();

  // Línea principal con suavizado
  ctx.strokeStyle = color;
  ctx.lineWidth = 2.5;
  ctx.lineJoin = 'round';
  ctx.lineCap = 'round';
  ctx.beginPath();
  
  datosFinales.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    
    if (i === 0) {
      ctx.moveTo(x, y);
    } else {
      ctx.lineTo(x, y);
    }
  });
  
  ctx.stroke();

  // Puntos en cada dato (más grandes con borde)
  const puntosCoords = [];
  datosFinales.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(x, y, 4, 0, Math.PI * 2);
    ctx.fill();
    
    // Borde para destacar
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 1.5;
    ctx.stroke();
    
    puntosCoords.push({ x, y, valor, label: labelsFinales[i] });
  });

  // Etiquetas X (solo algunas para no saturar)
  ctx.fillStyle = '#8b95a5';
  ctx.font = '10px system-ui';
  ctx.textAlign = 'center';
  
  const step = Math.max(1, Math.floor(labelsFinales.length / 6));
  labelsFinales.forEach((label, i) => {
    if (i % step === 0 || i === labelsFinales.length - 1) {
      const x = padding + i * stepX;
      // ✅ Manejar labels cortos de forma segura
      const displayLabel = label.length > 5 ? label.substring(5) : label;
      ctx.fillText(displayLabel, x, height - padding + 15);
    }
  });

  // ✅ Tooltip interactivo al pasar el mouse
  canvas.onmousemove = (e) => {
    const rect = canvas.getBoundingClientRect();
    const mouseX = e.clientX - rect.left;
    const mouseY = e.clientY - rect.top;
    
    let puntoCercano = null;
    let distanciaMin = 20;
    
    puntosCoords.forEach(punto => {
      const dist = Math.sqrt(
        Math.pow(mouseX - punto.x, 2) + 
        Math.pow(mouseY - punto.y, 2)
      );
      if (dist < distanciaMin) {
        distanciaMin = dist;
        puntoCercano = punto;
      }
    });
    
    if (puntoCercano) {
      canvas.style.cursor = 'pointer';
      canvas.title = `${puntoCercano.label}: $${puntoCercano.valor.toFixed(2)}`;
    } else {
      canvas.style.cursor = 'default';
      canvas.title = '';
    }
  };

  console.log(`📈 Gráfico de líneas dibujado: ${datosFinales.length} puntos`);
  return { canvas, ctx };
}

    .substring(0, maxLen);
}

function drawDoughnutChart(canvasId, labels, data, colors = ['#00e676', '#29b6f6', '#ab47bc', '#ffa726', '#66bb6a']) {
  const canvas = document.getElementById(canvasId);
  if (!canvas) {
    console.warn('⚠️ Canvas no encontrado:', canvasId);
    return null;
  }

  // ✅ Validación robusta de inputs
  if (!Array.isArray(labels)) labels = [];
  if (!Array.isArray(data)) data = [];
  
  // ✅ Filtrar datos inválidos (NaN, negativos)
  const datosLimpios = data.map(d => {
    const n = Number(d);
    return (isNaN(n) || n < 0) ? 0 : n;
  });
  
  // ✅ Asegurar misma longitud
  const maxLength = Math.max(labels.length, datosLimpios.length);
  const labelsFinales = Array.from({ length: maxLength }, (_, i) => 
    sanitizeForCanvas(labels[i] || `Categoría ${i + 1}`, 30)
  );
  const datosFinales = Array.from({ length: maxLength }, (_, i) => 
    datosLimpios[i] || 0
  );

  const ctx = canvas.getContext('2d');
  const width = canvas.width;
  const height = canvas.height;
  const centerX = width / 2;
  const centerY = height / 2;
  const radius = Math.min(width, height) / 2 - 40;
  const innerRadius = radius * 0.6;

  // ✅ Accesibilidad ARIA
  canvas.setAttribute('role', 'img');
  canvas.setAttribute('aria-label', `Gráfico de dona: ${labelsFinales.length} categorías`);

  // Limpiar canvas
  ctx.clearRect(0, 0, width, height);

  // ✅ Mensaje si no hay datos
  if (datosFinales.length === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('📭 Sin datos disponibles', centerX, centerY);
    return null;
  }

  const total = datosFinales.reduce((sum, val) => sum + val, 0);
  
  // ✅ Protección contra total = 0
  if (total === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('📊 Sin ventas aún', centerX, centerY);
    return null;
  }

  let startAngle = -Math.PI / 2;

  // ✅ Dibujar segmentos con validación
  datosFinales.forEach((valor, i) => {
    if (valor === 0) return; // No dibujar segmentos vacíos
    
    const sliceAngle = (valor / total) * 2 * Math.PI;
    const endAngle = startAngle + sliceAngle;

    ctx.fillStyle = colors[i % colors.length];
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius, startAngle, endAngle);
    ctx.arc(centerX, centerY, innerRadius, endAngle, startAngle, true);
    ctx.closePath();
    ctx.fill();

    // ✅ Borde sutil entre segmentos
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 2;
    ctx.stroke();

    startAngle = endAngle;
  });

  // ✅ Texto central con formato
  ctx.fillStyle = '#ffffff';
  ctx.font = 'bold 24px system-ui';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(total.toLocaleString(), centerX, centerY - 8);
  
  ctx.fillStyle = '#8b95a5';
  ctx.font = '11px system-ui';
  ctx.fillText('Total', centerX, centerY + 12);

  // ✅ Leyenda mejorada (máximo 6 categorías)
  const legendY = height - 20;
  const maxLabelsInLegend = Math.min(labelsFinales.length, 6);
  const legendSpacing = width / (maxLabelsInLegend + 1);
  
  ctx.font = '10px system-ui';
  labelsFinales.slice(0, maxLabelsInLegend).forEach((label, i) => {
    const x = legendSpacing * (i + 1);
    
    // ✅ Cuadro de color con borde
    ctx.fillStyle = colors[i % colors.length];
    ctx.fillRect(x - 30, legendY - 5, 10, 10);
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 1;
    ctx.strokeRect(x - 30, legendY - 5, 10, 10);
    
    // ✅ Texto (ya sanitizado)
    ctx.fillStyle = '#e0e0e0';
    ctx.textAlign = 'left';
    ctx.fillText(label, x - 15, legendY + 3);
  });

  // ✅ Indicador si hay más categorías
  if (labelsFinales.length > maxLabelsInLegend) {
    ctx.fillStyle = '#6a7585';
    ctx.font = '9px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText(`+${labelsFinales.length - maxLabelsInLegend} más`, width / 2, legendY + 15);
  }

  console.log(`🥧 Gráfico de dona dibujado: ${labelsFinales.length} categorías, total ${total}`);
  return { canvas, ctx };
}

async function exportFinancialData(formato = 'csv') {
  // ✅ Validar formato ANTES de procesar (whitelist)
  const formatosValidos = ['csv', 'json'];
  if (!formatosValidos.includes(formato)) {
    showNotification('❌ Formato no soportado. Usa CSV o JSON.', 'error');
    return;
  }

  // ✅ Verificar token
  const token = localStorage.getItem('token') || localStorage.getItem('auth_token');
  if (!token) {
    showNotification('⚠️ Debes iniciar sesión', 'warning');
    return;
  }

  try {
    showNotification('🔄 Preparando exportación...', 'info');
    
    const respuesta = await req('/api/analytics/financial?period=30d');
    
    if (!respuesta || !respuesta.success || !respuesta.data) {
      showNotification('❌ No hay datos para exportar', 'error');
      return;
    }

    const datos = respuesta.data;
    let contenido = '';
    let nombreArchivo = '';
    let tipoMIME = '';
    const fecha = new Date().toISOString().split('T')[0];

    if (formato === 'csv') {
      nombreArchivo = `analytics_${fecha}.csv`;
      tipoMIME = 'text/csv;charset=utf-8'; // ✅ UTF-8 para caracteres especiales
      
      // ✅ Función helper para escapar campos CSV
      const escapeCSV = (valor) => {
        if (valor === null || valor === undefined) return '';
        const str = String(valor);
        // Si contiene comas, comillas o saltos de línea, envolver en comillas
        if (str.includes(',') || str.includes('"') || str.includes('\n')) {
          return '"' + str.replace(/"/g, '""') + '"';
        }
        return str;
      };
      
      // Encabezado
      contenido = 'Fecha,Ingresos,Ventas\n';
      
      // Datos de ingresos por día
      if (Array.isArray(datos.revenueByDay)) {
        datos.revenueByDay.forEach(d => {
          const fecha = escapeCSV(d.date);
          const amount = Number(d.amount) || 0;
          const sales = Number(d.sales) || 0;
          contenido += `${fecha},${amount.toFixed(2)},${sales}\n`;
        });
      }
      
      contenido += '\n';
      
      // Datos de categorías
      contenido += 'Categoría,Cantidad\n';
      if (Array.isArray(datos.salesByCategory)) {
        datos.salesByCategory.forEach(c => {
          const categoria = escapeCSV(c.category);
          const count = Number(c.count) || 0;
          contenido += `${categoria},${count}\n`;
        });
      }
      
    } else if (formato === 'json') {
      nombreArchivo = `analytics_${fecha}.json`;
      tipoMIME = 'application/json;charset=utf-8';
      contenido = JSON.stringify(datos, null, 2);
    }

    // ✅ Validar que haya contenido
    if (!contenido) {
      showNotification('❌ No se pudo generar el archivo', 'error');
      return;
    }

    // Crear blob y descargar
    const blob = new Blob([contenido], { type: tipoMIME });
    const url = URL.createObjectURL(blob);
    const enlace = document.createElement('a');
    enlace.href = url;
    enlace.download = nombreArchivo;
    enlace.style.display = 'none';
    document.body.appendChild(enlace);
    enlace.click();
    document.body.removeChild(enlace);
    
    // ✅ Cleanup con delay
    setTimeout(() => URL.revokeObjectURL(url), 1000);

    showNotification(`✅ Archivo ${formato.toUpperCase()} descargado: ${nombreArchivo}`, 'success');
    log(`📤 Exportación completada: ${nombreArchivo}`, 'success');
    
  } catch (error) {
    const safeMsg = error.message || 'Error desconocido';
    log('❌ Error exportando datos: ' + safeMsg, 'error');
    showNotification('❌ Error al exportar: ' + safeMsg, 'error');
  }
}

// 📈 FUNCIÓN: DIBUJAR GRÁFICO DE LÍNEAS (MEJORADO)
// ==========================================
function dibujarGraficoLineas(canvasId, labels, datos, color = '#00e676') {
  const canvas = document.getElementById(canvasId);
  if (!canvas) {
    console.warn('⚠️ Canvas no encontrado:', canvasId);
    return;
  }

  // ✅ Validación robusta de entrada
  if (!Array.isArray(labels) || !Array.isArray(datos)) {
    console.warn('⚠️ labels o datos no son arrays');
    labels = [];
    datos = [];
  }

  // ✅ Filtrar datos inválidos (NaN, null, undefined)
  const datosLimpios = datos.map(d => {
    const n = Number(d);
    return isNaN(n) ? 0 : n;
  });

  // ✅ Asegurar que labels y datos tengan la misma longitud
  const maxLength = Math.max(labels.length, datosLimpios.length);
  const labelsFinal = Array.from({ length: maxLength }, (_, i) => 
    labels[i] || `Punto ${i + 1}`
  );
  const datosFinal = Array.from({ length: maxLength }, (_, i) => 
    datosLimpios[i] || 0
  );

  const ctx = canvas.getContext('2d');
  
  // ✅ Manejar redimensionamiento con devicePixelRatio
  const dpr = window.devicePixelRatio || 1;
  const displayWidth = canvas.offsetWidth || 600;
  const displayHeight = 300;
  
  canvas.width = displayWidth * dpr;
  canvas.height = displayHeight * dpr;
  canvas.style.width = displayWidth + 'px';
  canvas.style.height = displayHeight + 'px';
  ctx.scale(dpr, dpr);
  
  const width = displayWidth;
  const height = displayHeight;
  const padding = 40;

  ctx.clearRect(0, 0, width, height);

  // ✅ Mensaje si no hay datos
  if (datosFinal.length === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('📭 Sin datos disponibles', width / 2, height / 2);
    canvas.setAttribute('aria-label', 'Gráfico sin datos');
    return;
  }

  // ✅ Calcular rangos con protección contra NaN
  const maxVal = Math.max(...datosFinal);
  const minVal = Math.min(0, ...datosFinal);
  const rango = maxVal - minVal || 1;

  // ✅ Accesibilidad ARIA
  canvas.setAttribute('role', 'img');
  canvas.setAttribute('aria-label', 
    `Gráfico de líneas: ${datosFinal.length} puntos, máximo $${maxVal.toFixed(2)}, mínimo $${minVal.toFixed(2)}`
  );

  // Ejes
  ctx.strokeStyle = 'rgba(255,255,255,0.1)';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(padding, padding);
  ctx.lineTo(padding, height - padding);
  ctx.lineTo(width - padding, height - padding);
  ctx.stroke();

  // Grid horizontal con valores
  ctx.strokeStyle = 'rgba(255,255,255,0.05)';
  for (let i = 0; i <= 4; i++) {
    const y = padding + (height - 2 * padding) * (i / 4);
    ctx.beginPath();
    ctx.moveTo(padding, y);
    ctx.lineTo(width - padding, y);
    ctx.stroke();

    const valor = maxVal - (rango * i / 4);
    ctx.fillStyle = '#8b95a5';
    ctx.font = '10px system-ui';
    ctx.textAlign = 'right';
    ctx.fillText('$' + valor.toFixed(0), padding - 5, y + 3);
  }

  // ✅ Protección contra división por cero
  const stepX = datosFinal.length > 1 
    ? (width - 2 * padding) / (datosFinal.length - 1) 
    : 0;

  // Área bajo la línea con gradiente
  const gradient = ctx.createLinearGradient(0, padding, 0, height - padding);
  gradient.addColorStop(0, color + '40');
  gradient.addColorStop(1, color + '05');
  ctx.fillStyle = gradient;
  ctx.beginPath();
  ctx.moveTo(padding, height - padding);
  datosFinal.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minVal) / rango) * (height - 2 * padding);
    ctx.lineTo(x, y);
  });
  ctx.lineTo(padding + (datosFinal.length - 1) * stepX, height - padding);
  ctx.closePath();
  ctx.fill();

  // Línea principal con suavizado
  ctx.strokeStyle = color;
  ctx.lineWidth = 2.5;
  ctx.lineJoin = 'round';
  ctx.lineCap = 'round';
  ctx.beginPath();
  datosFinal.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minVal) / rango) * (height - 2 * padding);
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  });
  ctx.stroke();

  // Puntos con efecto hover (más grandes)
  ctx.fillStyle = color;
  const puntosCoords = [];
  datosFinal.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minVal) / rango) * (height - 2 * padding);
    ctx.beginPath();
    ctx.arc(x, y, 4, 0, Math.PI * 2);
    ctx.fill();
    // Borde blanco para destacar
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 1.5;
    ctx.stroke();
    puntosCoords.push({ x, y, valor, label: labelsFinal[i] });
  });

  // Etiquetas X (solo algunas para no saturar)
  ctx.fillStyle = '#8b95a5';
  ctx.font = '10px system-ui';
  ctx.textAlign = 'center';
  const step = Math.max(1, Math.floor(labelsFinal.length / 6));
  labelsFinal.forEach((label, i) => {
    if (i % step === 0 || i === labelsFinal.length - 1) {
      const x = padding + i * stepX;
      // ✅ Manejar labels cortos de forma segura
      const labelText = String(label);
      const displayLabel = labelText.length > 5 
        ? labelText.substring(5) 
        : labelText;
      ctx.fillText(displayLabel, x, height - padding + 15);
    }
  });

  // ✅ Tooltip interactivo al pasar el mouse
  canvas.onmousemove = (e) => {
    const rect = canvas.getBoundingClientRect();
    const mouseX = e.clientX - rect.left;
    const mouseY = e.clientY - rect.top;
    
    let puntoCercano = null;
    let distanciaMin = 20; // píxeles
    
    puntosCoords.forEach(punto => {
      const dist = Math.sqrt(
        Math.pow(mouseX - punto.x, 2) + 
        Math.pow(mouseY - punto.y, 2)
      );
      if (dist < distanciaMin) {
        distanciaMin = dist;
        puntoCercano = punto;
      }
    });
    
    if (puntoCercano) {
      canvas.style.cursor = 'pointer';
      canvas.title = `${puntoCercano.label}: $${puntoCercano.valor.toFixed(2)}`;
    } else {
      canvas.style.cursor = 'default';
      canvas.title = '';
    }
  };

  console.log(`📈 Gráfico de líneas dibujado: ${datosFinal.length} puntos`);
}

  );

  // ✅ Dibujar segmentos con animación
  let startAngle = -Math.PI / 2;
  const segmentos = []; // Guardar coords para tooltip

  datosFinal.forEach((valor, i) => {
    if (valor === 0) return; // No dibujar segmentos vacíos
    
    const sliceAngle = (valor / total) * 2 * Math.PI;
    const endAngle = startAngle + sliceAngle;

    ctx.fillStyle = colores[i % colores.length];
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius, startAngle, endAngle);
    ctx.arc(centerX, centerY, innerRadius, endAngle, startAngle, true);
    ctx.closePath();
    ctx.fill();

    // Borde sutil entre segmentos
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Guardar info del segmento para tooltip
    segmentos.push({
      startAngle,
      endAngle,
      label: labelsFinal[i],
      valor,
      porcentaje: ((valor / total) * 100).toFixed(1),
      color: colores[i % colores.length]
    });

    startAngle = endAngle;
  });

  // Texto central con total
  ctx.fillStyle = '#ffffff';
  ctx.font = 'bold 24px system-ui';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(total.toLocaleString(), centerX, centerY - 8);

  ctx.fillStyle = '#8b95a5';
  ctx.font = '11px system-ui';
  ctx.fillText('Total', centerX, centerY + 12);

  // ✅ Leyenda mejorada con manejo de muchas categorías
  const legendY = height - 25;
  const maxLabelsInLegend = Math.min(labelsFinal.length, 6); // Máximo 6 en leyenda
  const legendSpacing = width / (maxLabelsInLegend + 1);
  
  ctx.font = '10px system-ui';
  labelsFinal.slice(0, maxLabelsInLegend).forEach((label, i) => {
    const x = legendSpacing * (i + 1);
    
    // Cuadro de color
    ctx.fillStyle = colores[i % colores.length];
    ctx.fillRect(x - 30, legendY - 5, 10, 10);
    ctx.strokeStyle = '#0b0c10';
    ctx.lineWidth = 1;
    ctx.strokeRect(x - 30, legendY - 5, 10, 10);
    
    // Texto (truncar de forma segura)
    ctx.fillStyle = '#e0e0e0';
    ctx.textAlign = 'left';
    const labelText = String(label);
    const displayLabel = labelText.length > 10 
      ? labelText.substring(0, 9) + '…' 
      : labelText;
    ctx.fillText(displayLabel, x - 15, legendY + 3);
  });

  // ✅ Indicador si hay más categorías
  if (labelsFinal.length > maxLabelsInLegend) {
    ctx.fillStyle = '#6a7585';
    ctx.font = '9px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText(
      `+${labelsFinal.length - maxLabelsInLegend} más`, 
      width / 2, 
      legendY + 15
    );
  }

  // ✅ Tooltip interactivo al pasar el mouse
  canvas.onmousemove = (e) => {
    const rect = canvas.getBoundingClientRect();
    const mouseX = e.clientX - rect.left;
    const mouseY = e.clientY - rect.top;
    
    // Calcular ángulo y distancia desde el centro
    const dx = mouseX - centerX;
    const dy = mouseY - centerY;
    const distance = Math.sqrt(dx * dx + dy * dy);
    
    // Verificar si está en el anillo del dona
    if (distance >= innerRadius && distance <= radius) {
      let angle = Math.atan2(dy, dx);
      if (angle < -Math.PI / 2) angle += Math.PI * 2;
      
      // Encontrar el segmento correspondiente
      const segmento = segmentos.find(s => 
        angle >= s.startAngle && angle < s.endAngle
      );
      
      if (segmento) {
        canvas.style.cursor = 'pointer';
        canvas.title = `${segmento.label}: ${segmento.valor} (${segmento.porcentaje}%)`;
        return;
      }
    }
    
    canvas.style.cursor = 'default';
    canvas.title = '';
  };

  console.log(`🥧 Gráfico de dona dibujado: ${labelsFinal.length} categorías, total ${total}`);
}

if (typeof escapeHTML !== 'function') {
  function escapeHTML(texto) {
    if (texto === null || texto === undefined) return '';
    return String(texto)
      .replace(/&/g, '&')
      .replace(//g, '>')
      .replace(/"/g, '"')
      .replace(/'/g, ''');
  }
}

// 🛡️ HELPER: FORMATEAR MONEDA SEGURO
// ==========================================
function formatearMoneda(valor) {
  const num = Number(valor);
  if (isNaN(num)) return '$0.00';
  return '$' + num.toFixed(2);
}

// ==========================================
// 🛡️ HELPER: FORMATEAR FECHA SEGURO
// ==========================================
function formatearFecha(fecha) {
  if (!fecha) return 'Fecha desconocida';
  try {
    const d = new Date(fecha);
    if (isNaN(d.getTime())) return 'Fecha inválida';
    return d.toLocaleDateString();
  } catch (e) {
    return 'Fecha inválida';
  }
}

// 📥 FUNCIÓN: EXPORTAR DATOS FINANCIEROS (CORREGIDA)
// ==========================================
async function exportFinancialData(formato = 'csv') {
  // ✅ Validar formato antes de usar
  const formatosValidos = ['csv', 'json'];
  if (!formatosValidos.includes(formato)) {
    if (typeof showNotification === 'function') {
      showNotification('❌ Formato no soportado. Usa CSV o JSON.', 'error');
    } else {
      alert('❌ Formato no soportado. Usa CSV o JSON.');
    }
    return;
  }

  const token = localStorage.getItem('token') || localStorage.getItem('auth_token');
  if (!token) {
    if (typeof showNotification === 'function') {
      showNotification('⚠️ Debes iniciar sesión', 'warning');
    } else {
      alert('⚠️ Debes iniciar sesión');
    }
    return;
  }

  try {
    if (typeof showNotification === 'function') {
      showNotification('🔄 Preparando exportación...', 'info');
    }

    // ✅ RUTA RELATIVA + encodeURIComponent para sanitizar
    const respuesta = await fetch(
      `/api/analytics/financial/export?format=${encodeURIComponent(formato)}`,
      {
        method: 'GET',
        headers: {
          'Authorization': 'Bearer ' + token,
          'Content-Type': 'application/json'
        }
      }
    );

    // ✅ Manejo de sesión expirada
    if (respuesta.status === 401) {
      handleSessionExpired();
      return;
    }

    if (!respuesta.ok) throw new Error(`Error HTTP: ${respuesta.status}`);

    const datos = await respuesta.json();
    
    // ✅ Validación robusta
    if (!datos || !datos.success || !datos.data) {
      throw new Error('No hay datos para exportar');
    }

    let contenido = '';
    let nombreArchivo = '';
    let tipoMIME = '';
    const fecha = new Date().toISOString().split('T')[0];

    if (formato === 'csv') {
      contenido = datos.data.csv || '';
      nombreArchivo = `analytics_${fecha}.csv`;
      tipoMIME = 'text/csv;charset=utf-8';
    } else {
      contenido = JSON.stringify(datos.data.json || {}, null, 2);
      nombreArchivo = `analytics_${fecha}.json`;
      tipoMIME = 'application/json;charset=utf-8';
    }

    // ✅ Validar que haya contenido
    if (!contenido) {
      throw new Error('El servidor devolvió datos vacíos');
    }

    // ✅ Crear blob y descargar
    const blob = new Blob([contenido], { type: tipoMIME });
    const url = URL.createObjectURL(blob);
    const enlace = document.createElement('a');
    enlace.href = url;
    enlace.download = nombreArchivo;
    enlace.style.display = 'none';
    document.body.appendChild(enlace);
    enlace.click();
    document.body.removeChild(enlace);

    // ✅ Cleanup con delay para evitar problemas
    setTimeout(() => URL.revokeObjectURL(url), 1000);

    // ✅ Notificación no bloqueante
    if (typeof showNotification === 'function') {
      showNotification(`✅ Archivo ${formato.toUpperCase()} descargado`, 'success');
    } else {
      alert(`✅ Archivo ${formato.toUpperCase()} descargado`);
    }

    console.log(`📥 Exportación completada: ${nombreArchivo}`);
  } catch (error) {
    console.error('❌ Error exportando:', error);
    const msg = error.message || 'Error al exportar';
    if (typeof showNotification === 'function') {
      showNotification('❌ ' + msg, 'error');
    } else {
      alert('❌ Error al exportar: ' + msg);
    }
  }
}

  if (!token) {
    if (typeof showNotification === 'function') {
      showNotification('⚠️ Debes iniciar sesión', 'warning');
    } else {
      alert('⚠️ Debes iniciar sesión');
    }
    return;
  }

  const titulo = document.getElementById('mediaTitle');
  const contenido = document.getElementById('mediaContent');

      showNotification('❌ Error: No se encontró el visor multimedia', 'error');
    } else {
      alert('❌ Error: No se encontró el visor multimedia');
    }
    return;
  }

  modal.classList.remove('hidden');
  contenido.innerHTML = '
🔄 Desencriptando archivo...
';
  titulo.textContent = '📄 Cargando...';

  try {
    // ✅ RUTA RELATIVA (sin enlaces externos) + encodeURIComponent
    const respuesta = await fetch(`/vault/${encodeURIComponent(safeId)}`, {
      method: 'GET',
      headers: {
        'Authorization': 'Bearer ' + token,
        'Content-Type': 'application/json'
      }
    });

    // ✅ Manejo de sesión expirada
    if (respuesta.status === 401) {
      handleSessionExpired();
      return;
    }

    if (!respuesta.ok) throw new Error(`Error HTTP: ${respuesta.status}`);

    const datos = await respuesta.json();
    console.log('📦 Archivo recibido:', datos);

    // ✅ Validación robusta de respuesta
    if (!datos || typeof datos !== 'object') {
      throw new Error('Respuesta inválida del servidor');
    }
    if (!datos.success || !datos.secret) {
      throw new Error('No se pudo obtener el contenido');
    }

    const archivo = datos.secret;
    titulo.textContent = `📄 ${escapeHTML(archivo.titulo || 'Sin título')}`;

    const tipo = archivo.fileType || '';
    const contenidoBase64 = archivo.contenido;

    if (!contenidoBase64) {
      contenido.innerHTML = '
❌ Este archivo no tiene contenido descargable
';
      return;
    }

    // 🖼️ IMÁGENES
    if (tipo.startsWith('image/')) {
      contenido.innerHTML = `
        
      `;
    }
    // 🎬 VIDEOS
    else if (tipo.startsWith('video/')) {
      contenido.innerHTML = `
        
      `;
    }
    // 🎵 AUDIO
    else if (tipo.startsWith('audio/')) {
      contenido.innerHTML = `
        

          
🎵

          
          
${escapeHTML(archivo.fileName || archivo.titulo)}

        

      `;
    }
    // 📄 PDF
    else if (tipo === 'application/pdf') {
      contenido.innerHTML = `
        
      `;
    }
    // 📝 TEXTO
    else if (tipo.startsWith('text/') || tipo === 'application/json') {
      try {
        const texto = atob(contenidoBase64);
        contenido.innerHTML = `
          
${escapeHTML(texto)}

        `;
      } catch (e) {
        contenido.innerHTML = '
❌ No se pudo decodificar el texto
';
      }
    }
    // ❓ OTROS
    else {
      contenido.innerHTML = `
        

          
📦

          
Este tipo de archivo no tiene visor integrado

          
Tipo: ${escapeHTML(tipo)}

          
        

      `;
      // ✅ Usar addEventListener en lugar de onclick inline (más seguro)
      const btnDescargar = contenido.querySelector('[data-action="descargar"]');
      if (btnDescargar) {
        btnDescargar.addEventListener('click', () => descargarArchivo(safeId));
      }
    }
  } catch (error) {
    console.error('❌ Error:', error);
    let mensajeError = error.message;
    if (error.message.includes('Failed to fetch') || error.message.includes('NetworkError')) {
      mensajeError = 'Error de conexión. Verifica tu internet.';
    } else if (error.message.includes('404')) {
      mensajeError = 'Archivo no encontrado.';
    }
    contenido.innerHTML = `
❌ ${escapeHTML(mensajeError)}
`;
  }
}

// 📥 FUNCIÓN: DESCARGAR ARCHIVO (CORREGIDA)
// ==========================================
async function descargarArchivo(id) {
  // ✅ Sanitizar ID
  const safeId = sanitizeId(id);
  if (!safeId) {
    if (typeof showNotification === 'function') {
      showNotification('❌ ID de archivo inválido', 'error');
    } else {
      alert('❌ ID de archivo inválido');
    }
    return;
  }

  const token = localStorage.getItem('token') || localStorage.getItem('auth_token');
  if (!token) {
    if (typeof showNotification === 'function') {
      showNotification('⚠️ Debes iniciar sesión', 'warning');
    } else {
      alert('⚠️ Debes iniciar sesión');
    }
    return;
  }

  try {
    // ✅ RUTA RELATIVA (sin enlaces externos) + encodeURIComponent
    const respuesta = await fetch(`/vault/${encodeURIComponent(safeId)}`, {
      method: 'GET',
      headers: {
        'Authorization': 'Bearer ' + token,
        'Content-Type': 'application/json'
      }
    });

    // ✅ Manejo de sesión expirada
    if (respuesta.status === 401) {
      handleSessionExpired();
      return;
    }

    if (!respuesta.ok) throw new Error(`Error HTTP: ${respuesta.status}`);

    const datos = await respuesta.json();
    
    // ✅ Validación robusta
    if (!datos || !datos.success || !datos.secret || !datos.secret.contenido) {
      throw new Error('No se pudo obtener el contenido');
    }

    const archivo = datos.secret;
    const tipo = archivo.fileType || 'application/octet-stream';
    
    // ✅ Sanitizar nombre de archivo (prevenir ataques)
    let nombre = archivo.fileName || archivo.titulo || 'archivo';
    nombre = String(nombre)
      .replace(/[<>:"/\\|?*]/g, '_')  // Remover caracteres peligrosos
      .replace(/\s+/g, '_')           // Reemplazar espacios
      .substring(0, 100);             // Limitar longitud

    // ✅ Conversión base64 a blob
    const byteCharacters = atob(archivo.contenido);
    const byteNumbers = new Array(byteCharacters.length);
    for (let i = 0; i < byteCharacters.length; i++) {
      byteNumbers[i] = byteCharacters.charCodeAt(i);
    }
    const byteArray = new Uint8Array(byteNumbers);
    const blob = new Blob([byteArray], { type: tipo });

    // ✅ Crear enlace de descarga
    const url = URL.createObjectURL(blob);
    const enlace = document.createElement('a');
    enlace.href = url;
    enlace.download = nombre;
    enlace.style.display = 'none';
    document.body.appendChild(enlace);
    enlace.click();
    document.body.removeChild(enlace);

    // ✅ Cleanup con delay
    setTimeout(() => URL.revokeObjectURL(url), 1000);

    console.log('✅ Archivo descargado:', nombre);
    
    // ✅ Notificación no bloqueante
    if (typeof showNotification === 'function') {
      showNotification(`✅ Descargando: ${nombre}`, 'success');
    }
  } catch (error) {
    console.error('❌ Error descargando:', error);
    const msg = error.message || 'Error al descargar';
    if (typeof showNotification === 'function') {
      showNotification('❌ ' + msg, 'error');
    } else {
      alert('❌ ' + msg);
    }
  }
}

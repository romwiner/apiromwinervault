// ✅ Helper para sanitizar mensajes de error (previene XSS en logs)
function sanitizeErrorMessage(msg) {
  if (!msg) return '';
  return String(msg)
    .replace(/[&<>"']/g, (m) => ({'&':'&','<':'<','>':'>','"':'"',"'":'''}[m]))
    .substring(0, 300);
}

// ✅ Flag para evitar registrar listeners múltiples veces
let globalListenersInitialized = false;

async function init() {
  // Cargar preferencias Zero-Knowledge
  if (typeof loadZeroKnowledgePref === 'function') loadZeroKnowledgePref();
  
  // Verificar conexión API con retry
  let retries = 0;
  const maxRetries = 3;
  
  while (retries < maxRetries) {
    try {
      const s = await req('/api/status');
      const statusDot = document.getElementById('apiStatus');
      const statusText = document.getElementById('statusText');
      
      if (statusDot && statusText) {
        if (s.status === 'online' || s.success) {
          statusDot.className = 'status-dot online';
          statusText.textContent = `✅ Online • ${s.database || 'MongoDB'}`;
          log('🟢 API responde', 'success');
          showNotification('✅ Conectado al servidor', 'success', 3000);
        } else {
          statusDot.className = 'status-dot offline';
          statusText.textContent = '⚠️ Mantenimiento';
          log('⚠️ API en mantenimiento', 'warning');
        }
      }
      
      if (typeof checkPromo === 'function') checkPromo();
      break;
    } catch (e) {
      retries++;
      // ✅ Sanitizar mensaje de error antes de loguear (previene XSS)
      const safeMsg = sanitizeErrorMessage(e.message);
      log(`⚠️ Intento ${retries}/${maxRetries}: ${safeMsg}`, 'warning');
      
      if (retries === maxRetries) {
        const statusDot = document.getElementById('apiStatus');
        const statusText = document.getElementById('statusText');
        if (statusDot) statusDot.className = 'status-dot offline';
        if (statusText) statusText.textContent = '❌ Sin conexión';
        log('❌ API no responde tras 3 intentos', 'error');
        showNotification('❌ Error de conexión. Verifica tu internet.', 'error');
      } else {
        await new Promise(resolve => setTimeout(resolve, 1000 * retries));
      }
    }
  }
  
  // Cargar sesión si hay token
  if (token) {
    try {
      await loadUser();
      if (currentUser) {
        showDashboard();
        // Actualizar UI según tier
        updateTierUI();
      }
    } catch (e) {
      // ✅ Sanitizar mensaje de error
      const safeMsg = sanitizeErrorMessage(e.message);
      log('⚠️ Error cargando usuario: ' + safeMsg, 'warning');
      // No cerrar sesión automáticamente, permitir reintentar
    }
  }
  
  // Inicializar event listeners globales (solo una vez)
  setupGlobalListeners();
}

// 🎨 Actualizar UI según tier del usuario
function updateTierUI() {
  // Mostrar/ocultar funciones enterprise con ARIA para accesibilidad
  document.querySelectorAll('.enterprise-only').forEach(el => {
    const isVisible = userTier === 'enterprise';
    el.style.display = isVisible ? '' : 'none';
    el.setAttribute('aria-hidden', isVisible ? 'false' : 'true');
  });
  
  // Actualizar badge de tier
  const badge = document.getElementById('userTierBadge');
  if (badge) {
    // ✅ Sanitizar texto del badge (previene XSS si userTier viene de fuente externa)
    const safeTier = String(userTier || 'personal').replace(/[^a-z]/gi, '').substring(0, 20);
    badge.textContent = safeTier.charAt(0).toUpperCase() + safeTier.slice(1);
    badge.className = `tier-badge tier-${safeTier}`;
  }
  
  // Habilitar/deshabilitar funciones según quota
  if (userQuota) {
    const quotaDisplay = document.getElementById('quotaDisplay');
    const quotaFill = document.getElementById('quotaFill');
    const quotaText = document.getElementById('quotaText');
    
    if (quotaDisplay && quotaFill && quotaText) {
      const used = Number(userQuota.used || 0);
      const limit = Number(userQuota.limit || 0);
      
      // ✅ Prevenir división por cero
      const pct = limit > 0 ? Math.min(100, Math.max(0, (used / limit) * 100)) : 0;
      quotaFill.style.width = pct + '%';
      quotaFill.setAttribute('aria-valuenow', Math.round(pct));
      quotaText.textContent = `${used} / ${limit} items`;
      quotaDisplay.classList.remove('hidden');
      quotaDisplay.setAttribute('aria-hidden', 'false');
      
      // Alertas de cuota con mensajes sanitizados
      if (pct >= 90) {
        quotaText.className = 'quota-critical';
        quotaText.textContent += ' ⚠️ Casi lleno';
        quotaText.setAttribute('aria-label', 'Almacenamiento crítico: 90% o más usado');
      } else if (pct >= 75) {
        quotaText.className = 'quota-warning';
        quotaText.textContent += ' ⚠️ 75% usado';
        quotaText.setAttribute('aria-label', 'Almacenamiento: 75% o más usado');
      } else {
        quotaText.className = '';
        quotaText.setAttribute('aria-label', `Almacenamiento: ${Math.round(pct)}% usado`);
      }
    }
  }
}

// 🌐 Event listeners globales para accesibilidad y UX
function setupGlobalListeners() {
  // ✅ Prevenir registro múltiple de listeners (evita fugas de memoria)
  if (globalListenersInitialized) return;
  globalListenersInitialized = true;
  
  // Cerrar modales con tecla Escape
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') {
      // ✅ CORREGIDO: Seleccionar todos los modales, luego filtrar los que NO están ocultos
      document.querySelectorAll('.modal').forEach(modal => {
        if (!modal.classList.contains('hidden')) {
          modal.classList.add('hidden');
          modal.setAttribute('aria-hidden', 'true');
          log('🔒 Modal cerrado con Escape', 'info');
        }
      });
    }
  });
  
  // Cerrar modales al hacer clic fuera del contenido
  document.addEventListener('click', (e) => {
    document.querySelectorAll('.modal').forEach(modal => {
      if (!modal.classList.contains('hidden') && e.target === modal) {
        modal.classList.add('hidden');
        modal.setAttribute('aria-hidden', 'true');
        log('🔒 Modal cerrado al hacer clic fuera', 'info');
      }
    });
  });
}
  // Cerrar modales al hacer clic fuera del contenido
  document.querySelectorAll('.modal').forEach(modal => {
    modal.addEventListener('click', (e) => {
      if (e.target === modal) {
        modal.classList.add('hidden');
        modal.setAttribute('aria-hidden', 'true');
        log('🔒 Modal cerrado al hacer clic fuera', 'info');
      }
    });
  });
  
  // Prevenir scroll cuando hay modal abierto
  let modalObserver = null;
  const setupModalObserver = () => {
    if (modalObserver) return; // ✅ Evitar múltiples observers
    modalObserver = new MutationObserver(() => {
      const hasOpenModal = Array.from(document.querySelectorAll('.modal')).some(m => !m.classList.contains('hidden'));
      document.body.style.overflow = hasOpenModal ? 'hidden' : '';
    });
    modalObserver.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class'] });
  };
  setupModalObserver();
  
  // ✅ Limpieza de observer al cerrar la página (previene fugas de memoria)
  window.addEventListener('beforeunload', () => {
    if (modalObserver) modalObserver.disconnect();
  });
}

  const forSale = document.getElementById('uploadForSale');
  const priceSection = document.getElementById('priceSection');
  const priceInput = document.getElementById('uploadPrice');
  const pricePreview = document.getElementById('pricePreview');

      priceSection.style.display = forSale.checked ? 'block' : 'none';

    }
  }
});

// 📤 Subir archivo con progreso real y seguridad mejorada
async function doUpload() {
  // ✅ Prevenir múltiples uploads simultáneos
  if (isUploading) {
    showNotification('⚠️ Ya hay una subida en progreso', 'warning');
    return;
  }
  
  const titleInput = document.getElementById('uploadTitle');
  const catSelect = document.getElementById('uploadCat');
  const fileInput = document.getElementById('uploadFile');
  const forSaleCheck = document.getElementById('uploadForSale');
  const priceInput = document.getElementById('uploadPrice');
  const status = document.getElementById('uploadStatus');
  const btn = document.getElementById('uploadBtn');
  const progress = document.getElementById('uploadProgress');
  const progressFill = document.getElementById('uploadProgressFill');
  
  const title = titleInput?.value?.trim();
  const cat = catSelect?.value;
  const file = fileInput?.files?.[0];
  const forSale = !!forSaleCheck?.checked;
  const price = parseFloat(priceInput?.value) || 0;
  
  // Validaciones con feedback accesible
  if (!title || title.length < 3) {
    showNotification('⚠️ El título debe tener al menos 3 caracteres', 'error');
    if (titleInput) {
      titleInput.classList.add('error');
      titleInput.setAttribute('aria-invalid', 'true');
      titleInput.focus();
      setTimeout(() => {
        titleInput.classList.remove('error');
        titleInput.removeAttribute('aria-invalid');
      }, 2000);
    }
    return;
  }
  
  if (!file) {
    showNotification('⚠️ Selecciona un archivo para subir', 'error');
    if (fileInput) {
      fileInput.classList.add('error');
      fileInput.focus();
      setTimeout(() => fileInput.classList.remove('error'), 2000);
    }
    return;
  }
  
  const fileError = validateFile(file);
  if (fileError) {
    showNotification(fileError, 'error');
    return;
  }
  
  if (forSale && (isNaN(price) || price < 1 || price > 9999)) {
    showNotification('⚠️ El precio debe estar entre $1.00 y $9,999.00 USD', 'error');
    if (priceInput) {
      priceInput.classList.add('error');
      priceInput.focus();
      setTimeout(() => priceInput.classList.remove('error'), 2000);
    }
    return;
  }
  
  // ✅ Marcar como subiendo
  isUploading = true;
  
  // UI: estado de carga con accesibilidad
  btn.disabled = true;
  btn.textContent = '⏳ Cifrando y subiendo...';
  btn.setAttribute('aria-busy', 'true');
  if (status) {
    status.textContent = '🔐 Cifrando archivo...';
    status.setAttribute('aria-live', 'polite');
  }
  if (progress) progress.classList.remove('hidden');
  
  try {
    // Demo: simular progreso con cleanup garantizado
    let pct = 0;
    let interval = null;
    const startProgress = () => {
      interval = setInterval(() => {
        pct = Math.min(95, pct + Math.random() * 8 + 2);
        if (progressFill) {
          progressFill.style.width = Math.round(pct) + '%';
          progressFill.setAttribute('aria-valuenow', Math.round(pct));
        }
      }, 250);
    };
    const stopProgress = () => {
      if (interval) {
        clearInterval(interval);
        interval = null;
      }
    };
    
    startProgress();
    
    // En producción: FormData + fetch con onUploadProgress
    const formData = new FormData();
    formData.append('file', file);
    formData.append('titulo', title);
    formData.append('categoria', cat);
    formData.append('forSale', forSale.toString());
    if (forSale) formData.append('price', price.toFixed(2));
    
    // Simular llamada al backend
    await new Promise(resolve => setTimeout(resolve, 2000));
    
    stopProgress();
    
    // ✅ Completar progreso visual
    if (progressFill) {
      progressFill.style.width = '100%';
      progressFill.setAttribute('aria-valuenow', '100');
    }
    
    // Respuesta demo
    const response = {
      success: true,
      message: '✅ Archivo subido y cifrado exitosamente',
      item: { id: 'demo_' + Date.now(), titulo: title, categoria: cat }
    };
    
    if (status) {
      status.textContent = response.message;
      status.style.color = '#00e676';
    }
    
    showNotification(response.message, 'success');
    
    // Limpiar formulario y cerrar
    if (titleInput) titleInput.value = '';
    if (fileInput) fileInput.value = '';
    if (priceInput) priceInput.value = '';
    if (forSaleCheck) forSaleCheck.checked = false;
    if (priceSection) priceSection.style.display = 'none';
    
    setTimeout(() => {
      closeUpload();
      if (typeof refresh === 'function') refresh();
    }, 1500);
    
  } catch (e) {
    // ✅ Mensaje de error seguro: no exponer detalles internos
    const safeError = 'Error al subir archivo';
    if (status) {
      status.textContent = `❌ ${safeError}`;
      status.style.color = '#ff5252';
    }
    showNotification(`❌ ${safeError}`, 'error');
    console.error('Upload error:', { name: e.name, message: e.message });
    
  } finally {
    // ✅ Cleanup garantizado
    isUploading = false;
    if (btn) {
      btn.disabled = false;
      btn.textContent = '✅ Subir y Cifrar';
      btn.removeAttribute('aria-busy');
    }
    setTimeout(() => {
      if (progress) progress.classList.add('hidden');
      if (progressFill) {
        progressFill.style.width = '0%';
        progressFill.setAttribute('aria-valuenow', '0');
      }
    }, 1000);

// === DASHBOARD: REFRESH MEJORADO CON SEGURIDAD ===
async function refresh(event) {
  // ✅ Prevenir comportamiento por defecto si es submit
  if (event) event.preventDefault();
  
  // ✅ Verificar token antes de continuar
  if (!token) {
    log('⚠️ Sin token, no se puede refrescar', 'warning');
    showNotification('⚠️ Inicia sesión para actualizar', 'warning');
    return;
  }
  
  // ✅ Referencia segura al botón con fallback
  const btn = event?.currentTarget || document.querySelector('button[onclick="refresh()"]');
  
  try {
    // UI: estado de carga con accesibilidad
    if (btn) {
      btn.disabled = true;
      btn.textContent = '🔄 Actualizando...';
      btn.setAttribute('aria-busy', 'true');
    }
    
    // ✅ Fetch paralelo con AbortController para evitar bloqueos infinitos
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 15000); // 15s timeout
    
    const [w, d, a] = await Promise.all([
      req('/api/wallet', 'GET', null, false, { signal: controller.signal }),
      req('/api/dashboard', 'GET', null, false, { signal: controller.signal }),
      req('/api/affiliates/dashboard', 'GET', null, false, { signal: controller.signal })
    ]);

    const saldoEl = document.getElementById('saldo');
    if (saldoEl) {
      const balance = Number(w.wallet?.balance || 0);
      saldoEl.textContent = fmt(balance);
      saldoEl.setAttribute('aria-label', `Saldo disponible: ${fmt(balance)}`);

    if (activosEl) {
      const count = Number(d.dashboard?.active || 0);
      activosEl.textContent = count;
      activosEl.setAttribute('aria-label', `${count} activos en venta`);
    }

    if (ventasEl) {
      const count = Number(d.dashboard?.sales || 0);
      ventasEl.textContent = count;
      ventasEl.setAttribute('aria-label', `${count} ventas completadas`);
    }
    
    const afiliadosEl = document.getElementById('afiliados');
    if (afiliadosEl) {
      const balance = Number(a.dashboard?.availableBalance || 0);
      afiliadosEl.textContent = fmt(balance);
      afiliadosEl.setAttribute('aria-label', `Ganancias de afiliados: ${fmt(balance)}`);
    }

    if (d.dashboard?.quota) {
      userQuota = d.dashboard.quota;
      if (typeof updateQuotaDisplay === 'function') updateQuotaDisplay();
    }
    
    log('🟢 Métricas actualizadas', 'success');
    
  } catch (e) {
    // ✅ Manejo específico de errores para mejor UX
    if (e.name === 'AbortError') {
      log('⚠️ Timeout actualizando métricas', 'warning');
      showNotification('⚠️ La conexión es lenta. Intenta de nuevo.', 'warning');
    } else if (e.message.includes('401')) {
      log('⚠️ Sesión expirada', 'warning');
      showNotification('⚠️ Sesión expirada. Inicia sesión de nuevo.', 'warning');
    } else if (e.message.includes('403')) {
      log('⚠️ Acceso denegado', 'warning');
      showNotification('⚠️ No tienes permiso para esta acción.', 'warning');
    } else {
      log('⚠️ Error sincronizando: ' + sanitizeForDisplay(e.message, 100), 'warning');
      showNotification('⚠️ No se pudo actualizar. Verifica tu conexión.', 'warning');
    }
  } finally {
    // ✅ Restaurar UI siempre, incluyendo atributos ARIA
    if (btn) {
      btn.disabled = false;
      btn.textContent = '🔄 Actualizar Métricas';
      btn.removeAttribute('aria-busy');
    }
  }
}

// === QUOTA: DISPLAY MEJORADO CON ACCESIBILIDAD ===
function updateQuotaDisplay() {
  if (!userQuota) return;
  
  // ✅ Null checks seguros para todos los elementos DOM
  const display = document.getElementById('quotaDisplay');
  const fill = document.getElementById('quotaFill');
  const text = document.getElementById('quotaText');
  
  if (!display || !fill || !text) {
    log('⚠️ Elementos de quota no encontrados', 'warning');
    return;
  }
  
  const used = Number(userQuota.usedGB || 0);
  const limit = userQuota.tier?.storageLimitGB;
  
  // ✅ Caso: almacenamiento ilimitado
  if (limit === -1 || limit === Infinity) {
    text.textContent = '📦 Almacenamiento ilimitado';
    text.className = '';
    text.setAttribute('aria-label', 'Tu plan tiene almacenamiento ilimitado');
    fill.style.width = '100%';
    fill.setAttribute('aria-valuenow', '100');
    display.classList.remove('hidden');
    display.setAttribute('aria-hidden', 'false');
    return;
  }
  
  // ✅ Caso: límite definido
  if (limit && limit > 0) {
    const percent = Math.min(100, Math.max(0, (used / limit) * 100));
    const percentRounded = Math.round(percent);
    
    // Actualizar barra de progreso con atributos ARIA
    fill.style.width = percentRounded + '%';
    fill.setAttribute('aria-valuenow', percentRounded);
    fill.setAttribute('aria-valuemin', '0');
    fill.setAttribute('aria-valuemax', '100');
    
    // Actualizar texto con formato consistente
    text.textContent = `${used.toFixed(2)} GB / ${limit} GB`;
    
    // ✅ Alertas visuales y de accesibilidad según uso
    if (percent >= 90) {
      text.className = 'quota-critical';
      text.textContent += ' ⚠️ Casi lleno';
      text.setAttribute('aria-label', `Almacenamiento crítico: ${percentRounded}% usado`);
      // ✅ Notificación única para no saturar
      if (!display.dataset.criticalNotified) {
        showNotification('⚠️ Tu almacenamiento está casi lleno (90%)', 'warning', 8000);
        display.dataset.criticalNotified = 'true';
      }
    } else if (percent >= 75) {
      text.className = 'quota-warning';
      text.textContent += ' ⚠️ Considera limpiar';
      text.setAttribute('aria-label', `Almacenamiento: ${percentRounded}% usado`);
      delete display.dataset.criticalNotified;
    } else {
      text.className = '';
      text.setAttribute('aria-label', `Almacenamiento: ${percentRounded}% usado`);
      delete display.dataset.criticalNotified;
    }
    
    display.classList.remove('hidden');
    display.setAttribute('aria-hidden', 'false');
    
  } else {
    // ✅ Caso: sin datos de quota
    display.classList.add('hidden');
    display.setAttribute('aria-hidden', 'true');
  }
}

  return String(str)

    .replace(/"/g, '"')
    .substring(0, maxLen);
}

// 🔐 Simular cifrado Envelope (local, sin conexiones externas)
async function simulateEnvelopeEncryption(titulo, categoria) {
  // ✅ En producción: usar WebCrypto API para cifrado real
  // Aquí simulamos con hash + timestamp para demo segura
  const data = `${titulo}|${categoria}|${Date.now()}|${Math.random().toString(36).slice(2)}`;
  
  // Hash simple (SHA-256 simulado para demo)
  let hash = 0;
  for (let i = 0; i < data.length; i++) {
    const char = data.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash = hash & hash;
  }
  
  // Retornar "contenido cifrado" simulado
  return `ENC:v1:${Math.abs(hash).toString(16)}:${btoa(data).slice(0, 50)}...`;
}

  if (progress) progress.classList.remove('hidden');
  if (progressFill) {
    progressFill.style.width = '0%';
    progressFill.setAttribute('aria-valuenow', '0');
  }
  if (status) {
    status.textContent = '🔐 Cifrando archivo...';

    status.setAttribute('aria-live', 'polite');
  }
  if (btn) {
    btn.disabled = true;
    btn.textContent = '⏳ Subiendo...';
    btn.setAttribute('aria-busy', 'true');
  }
  
  // ✅ Control de progreso con cleanup garantizado
  let prog = 0;
  let interval = null;
  const startProgress = () => {
    interval = setInterval(() => {
      prog = Math.min(90, prog + Math.random() * 8 + 2); // Progreso realista
      if (progressFill) {
        progressFill.style.width = Math.round(prog) + '%';
        progressFill.setAttribute('aria-valuenow', Math.round(prog));
      }
    }, 250);
  };
  
  const stopProgress = () => {
    if (interval) {
      clearInterval(interval);
      interval = null;
    }
  };
  
  try {
    // ✅ Iniciar progreso simulado
    startProgress();
    
    // ✅ Preparar FormData con validación
    const formData = new FormData();
    formData.append('titulo', titulo);
    formData.append('categoria', cat);
    formData.append('forSale', forSale.toString());
    if (forSale) formData.append('price', price.toFixed(2));
    formData.append('archivo', file, file.name); // ✅ Nombre original preservado

    // ✅ Fetch con AbortController para cancelación

    // ✅ Enviar al backend
    const r = await fetch(API + '/vault', {
      method: 'POST',
      headers,
      body: formData,
      signal: controller.signal,
      credentials: 'same-origin'
    });
    
    clearTimeout(timeoutId);
    stopProgress();
    
    // ✅ Procesar respuesta

      progressFill.style.width = '100%';
      progressFill.setAttribute('aria-valuenow', '100');
    }

    }

      status.textContent = '✅ Archivo subido y cifrado';
      status.style.color = '#00e676';

    // ✅ Reset y refresh con delay para feedback visual
    setTimeout(() => {
      closeUpload();
      if (typeof refresh === 'function') refresh();
      if (typeof showVault === 'function') showVault();
    }, 1200);
    
  } catch (e) {
    stopProgress();
    
    // ✅ Resetear progreso visual
    if (progressFill) {
      progressFill.style.width = '0%';
      progressFill.setAttribute('aria-valuenow', '0');
    }
    
    // ✅ Mensajes de error específicos
    let errorMsg = e.message;
    if (e.name === 'AbortError') {
      errorMsg = '⏱️ La subida tardó demasiado. Verifica tu conexión.';
    } else if (errorMsg.includes('401')) {
      errorMsg = '❌ Sesión expirada. Inicia sesión de nuevo.';
    } else if (errorMsg.includes('413')) {
      errorMsg = '⚠️ Archivo demasiado grande para el servidor.';
    } else if (errorMsg.includes('400')) {
      errorMsg = '❌ Datos inválidos. Verifica el formulario.';
    }
    
    if (status) {
      status.textContent = '❌ Error: ' + sanitizeForDisplay(errorMsg, 100);
      status.style.color = '#ff5252';
    }
    
    log('❌ Upload fallido: ' + sanitizeForDisplay(e.message, 100), 'error');
    showNotification(sanitizeForDisplay(errorMsg, 100), 'error');
    console.error('Upload error details:', { name: e.name, message: e.message, stack: e.stack });
    
  } finally {
    // ✅ Cleanup garantizado: siempre restaurar UI
    isUploading = false;
    stopProgress(); // ✅ Asegurar que el intervalo se limpie
    
    if (btn) {
      btn.disabled = false;
      btn.textContent = '✅ Subir y Cifrar';
      btn.removeAttribute('aria-busy');
    }
  }
}
// ✅ Helper para sanitizar texto antes de insertar en DOM/logs (previene XSS)
function sanitizeForDisplay(str, maxLen = 100) {
  if (!str) return '';
  return String(str)
    .replace(/[&<>"']/g, (m) => ({'&':'&','<':'<','>':'>','"':'"',"'":'''}[m]))
    .substring(0, maxLen);
}

// ✅ Helper para sanitizar IDs usados en onclick (previene inyección de código)
function sanitizeIdForEvent(str) {
  if (!str) return '';
  return String(str).replace(/[^a-zA-Z0-9\-_]/g, '').substring(0, 50);
}

// ✅ Compartir archivo con sanitización y validación
async function shareFile(fileId) {
  // ✅ Sanitizar ID antes de usar
  const safeId = sanitizeIdForEvent(fileId);
  if (!safeId) {
    showNotification('⚠️ ID de archivo inválido', 'error');
    return;
  }
  
  try {
    const pass = prompt('🔒 Contraseña opcional (Enter para ninguna):');
    // ✅ Validar expiración: número entre 1 y 168 horas (1 semana)
    const hoursInput = prompt('⏱️ Expirar en horas (1-168):', '24') || '24';
    const hours = parseInt(hoursInput, 10);
    if (isNaN(hours) || hours < 1 || hours > 168) {
      showNotification('⚠️ Expiración debe estar entre 1 y 168 horas', 'error');
      return;
    }
    
    const res = await req(`/api/vault/${safeId}/share`, 'POST', {
      password: pass || null,
      expiresInHours: hours
    });
    
    // ✅ Copiar al portapapeles con fallback
    if (navigator.clipboard && res.link) {
      await navigator.clipboard.writeText(res.link);
      showNotification('✅ Enlace copiado al portapapeles', 'success');
    } else if (res.link) {
      // Fallback para navegadores antiguos
      const textarea = document.createElement('textarea');
      textarea.value = res.link;
      document.body.appendChild(textarea);
      textarea.select();
      document.execCommand('copy');
      document.body.removeChild(textarea);
      showNotification('✅ Enlace copiado', 'success');
    }
    
    log(`🔗 Share generado para: ${safeId}`, 'info');
    
  } catch (e) {
    // ✅ Mensaje seguro: no exponer detalles internos
    const safeMsg = 'Error al generar enlace de compartir';
    log('❌ Share: ' + sanitizeForDisplay(e.message, 100), 'error');
    showNotification(`❌ ${safeMsg}`, 'error');
  }
}

    let exportData;

      throw new Error('El archivo no contiene JSON válido');
    }
    
    status.textContent = '🔄 Importando...';

      exportData, 
      options: { forceImport: force } 
    });

    const imported = Number(res.summary?.imported || 0);
    const errors = Number(res.summary?.errors || 0);

    if (btn) {
      btn.disabled = false;
      btn.textContent = '✅ Importar';
      btn.removeAttribute('aria-busy');
    }
  }
}

// ✅ FUNCIÓN req() CORREGIDA - Acepta ambos formatos de llamada
async function req(url, metodoOpciones, bodyOpcional) {
  try {
    let opciones = {};
    
    // Soporta ambos formatos:
    // Formato 1: req(url, { method: 'POST', body: {...} })
    // Formato 2: req(url, 'POST', { message: '...' })
    if (typeof metodoOpciones === 'string') {
      opciones = {
        method: metodoOpciones,
        body: bodyOpcional ? JSON.stringify(bodyOpcional) : undefined
      };
    } else if (typeof metodoOpciones === 'object') {
      opciones = {
        method: metodoOpciones.method || 'GET',
        body: metodoOpciones.body ? JSON.stringify(metodoOpciones.body) : undefined
      };
    }

    const respuesta = await fetch(url, {
      method: opciones.method || 'GET',
      headers: { 
        'Content-Type': 'application/json',
        ...(opciones.headers || {})
      },
      body: opciones.body
    });

    if (!respuesta.ok) {
      const textoError = await respuesta.text().catch(() => '');
      throw new Error(`HTTP ${respuesta.status}: ${respuesta.statusText} ${textoError}`);
    }

    const texto = await respuesta.text();
    return texto ? JSON.parse(texto) : {};
  } catch (error) {
    log(`❌ Error en ${url}: ${error.message}`, 'error');
    throw error;
  }
}

// ✅ NUEVA FUNCIÓN: Sanitizar texto para mostrar en pantalla (evita XSS)
function sanitizeForDisplay(texto, maxLongitud = 100) {
  if (texto === null || texto === undefined) return '';
  let limpio = String(texto);
  // Escapar caracteres HTML peligrosos
  limpio = limpio
    .replace(/&/g, '&')
    .replace(//g, '>')
    .replace(/"/g, '"')
    .replace(/'/g, ''');
  // Limitar longitud
  if (maxLongitud && limpio.length > maxLongitud) {
    limpio = limpio.substring(0, maxLongitud) + '...';
  }
  return limpio;
}

// Inicializar canvas cuando el DOM esté listo
function initSignatureCanvas() {
  if (!sigCanvas) {
    log('⚠️ Canvas de firma no encontrado', 'warning');
    return;
  }

  sigContext = sigCanvas.getContext('2d');
  
  // ✅ Configurar estilo del trazo
  sigContext.strokeStyle = '#ffffff';
  sigContext.lineWidth = 2.5;
  sigContext.lineCap = 'round';
  sigContext.lineJoin = 'round';

  // ✅ Función para obtener posición del cursor/tacto
  function getPosition(evento) {
    const rect = sigCanvas.getBoundingClientRect();
    const clienteX = evento.touches?.[0]?.clientX || evento.clientX;
    const clienteY = evento.touches?.[0]?.clientY || evento.clientY;
    return {
      x: clienteX - rect.left,
      y: clienteY - rect.top
    };
  }

  // ✅ Usar addEventListener (no sobrescribe otros handlers)
  sigCanvas.addEventListener('mousedown', (e) => {
    isDrawing = true;
    const pos = getPosition(e);
    sigContext.beginPath();
    sigContext.moveTo(pos.x, pos.y);
  });

  sigCanvas.addEventListener('mousemove', (e) => {
    if (!isDrawing) return;
    const pos = getPosition(e);
    sigContext.lineTo(pos.x, pos.y);
    sigContext.stroke();
  });

  sigCanvas.addEventListener('mouseup', () => {
    isDrawing = false;
  });

  sigCanvas.addEventListener('mouseleave', () => {
    isDrawing = false;
  });

  // Soporte táctil
  sigCanvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    isDrawing = true;
    const pos = getPosition(e);
    sigContext.beginPath();
    sigContext.moveTo(pos.x, pos.y);
  }, { passive: false });

  sigCanvas.addEventListener('touchmove', (e) => {
    e.preventDefault();
    if (!isDrawing) return;
    const pos = getPosition(e);
    sigContext.lineTo(pos.x, pos.y);
    sigContext.stroke();
  }, { passive: false });

  sigCanvas.addEventListener('touchend', () => {
    isDrawing = false;
  });

  log('✅ Canvas de firma inicializado', 'success');
}

  if (archivo.size > tamanoMaximo) {
    showNotification('❌ La imagen es muy grande (máx. 5MB)', 'error');
    evento.target.value = '';
    return;
  }

  const avatarImg = document.getElementById('userAvatar');
  if (!avatarImg) {
    showNotification('❌ Elemento de avatar no encontrado', 'error');
    return;
  }

  // Leer y guardar la imagen
  const reader = new FileReader();
  reader.onload = (e) => {
    avatarImg.src = e.target.result;

      localStorage.setItem('romwiner_avatar', e.target.result);

    }
  };
  
  reader.onerror = () => {
    showNotification('❌ Error al leer la imagen', 'error');
  };
  
  reader.readAsDataURL(archivo);
}

// 🔐 FUNCIÓN HASH SHA-256 NATIVO (CORREGIDA)
// ============================================
async function hashLocal(data, longitudCompleta = false) {
  if (!data) return 'none';
  
  // Verificar que crypto.subtle esté disponible
  if (typeof crypto === 'undefined' || !crypto.subtle) {
    log('⚠️ crypto.subtle no disponible (requiere HTTPS)', 'warning');
    return 'none';
  }

  try {
    const encoder = new TextEncoder();
    const buffer = encoder.encode(String(data));
    const hashBuffer = await crypto.subtle.digest('SHA-256', buffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    
    // ✅ Devolver hash completo o truncado según se solicite
    return longitudCompleta ? hashHex : hashHex.slice(0, 16);
  } catch (error) {
    log('⚠️ Error generando hash: ' + error.message, 'warning');
    return 'none';
  }
}

async function generateIdenticon(elementoId, datos) {
  const elemento = document.getElementById(elementoId);
  if (!elemento) {
    log(`⚠️ Elemento #${elementoId} no encontrado`, 'warning');
    return;
  }

  try {
    // Generar hash de los datos para crear el patrón único
    const hashCompleto = await hashLocal(JSON.stringify(datos), true);
    
    if (hashCompleto === 'none') {
      log('⚠️ No se pudo generar hash para identicon', 'warning');
      return;
    }

    const tamano = 150;
    const celdas = 5; // Grid de 5x5 (simétrico)
    const tamanoCelda = tamano / celdas;
    
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = tamano;
    const ctx = canvas.getContext('2d');

    // Fondo oscuro
    ctx.fillStyle = '#0b0c10';
    ctx.fillRect(0, 0, tamano, tamano);

    // Extraer colores del hash
    const colorHex = '#' + hashCompleto.slice(0, 6);
    const colorSecundario = '#' + hashCompleto.slice(6, 12);

    // Generar patrón simétrico (espejo horizontal)
    for (let fila = 0; fila < celdas; fila++) {
      for (let columna = 0; columna < Math.ceil(celdas / 2); columna++) {
        // Usar caracteres del hash para decidir si pintar
        const indice = (fila * celdas + columna) % hashCompleto.length;
        const valor = parseInt(hashCompleto[indice], 16);
        
        if (valor > 7) { // 50% de probabilidad
          // Color según la posición
          ctx.fillStyle = (fila + columna) % 2 === 0 ? colorHex : colorSecundario;
          
          // Pintar celda y su espejo
          ctx.fillRect(columna * tamanoCelda, fila * tamanoCelda, tamanoCelda - 1, tamanoCelda - 1);
          ctx.fillRect((celdas - 1 - columna) * tamanoCelda, fila * tamanoCelda, tamanoCelda - 1, tamanoCelda - 1);
        }
      }
    }

    // Borde sutil
    ctx.strokeStyle = '#2a2f3a';
    ctx.lineWidth = 2;
    ctx.strokeRect(0, 0, tamano, tamano);

    // Convertir a imagen y mostrar
    elemento.src = canvas.toDataURL();
    elemento.alt = 'Identicon de identidad';
    elemento.style.borderRadius = '8px';
    
    log('🎨 Identicon generado (100% local, único para este usuario)', 'success');
    
  } catch (error) {
    log('❌ Error generando identicon: ' + error.message, 'error');
  }
}

// Módulo QR mínimo - Implementación compacta
const QRModule = (function() {
  'use strict';

  // Tablas de corrección de errores (simplificadas para versión 1-2)
  const EC_CODEWORDS_PER_BLOCK = {
    1: { L: 7, M: 10, Q: 13, H: 17 },
    2: { L: 10, M: 16, Q: 22, H: 28 },
    3: { L: 15, M: 26, Q: 36, H: 44 }
  };

  // Capacidades en bytes (modo byte, nivel L)
  const CAPACITIES = {
    1: 17, 2: 32, 3: 53, 4: 78, 5: 106,
    6: 134, 7: 154, 8: 192, 9: 230, 10: 271
  };

  // Determinar versión necesaria
  function getVersion(dataLength) {
    for (let v = 1; v <= 10; v++) {
      if (CAPACITIES[v] >= dataLength) return v;
    }
    return -1; // Datos muy grandes
  }

  // Codificar datos en modo byte
  function encodeData(text, version) {
    const bytes = [];
    // Modo byte: 0100 (4 bits)
    // Longitud: 8 bits (versión 1-9) o 16 bits (versión 10+)
    // Datos: 8 bits por carácter
    
    const dataBits = [];
    
    // Indicador de modo (0100 = byte)
    dataBits.push(0, 1, 0, 0);
    
    // Longitud
    const len = text.length;
    const lenBits = version <= 9 ? 8 : 16;
    for (let i = lenBits - 1; i >= 0; i--) {
      dataBits.push((len >> i) & 1);
    }
    
    // Datos
    for (let i = 0; i < text.length; i++) {
      const code = text.charCodeAt(i) & 0xFF;
      for (let j = 7; j >= 0; j--) {
        dataBits.push((code >> j) & 1);
      }
    }
    
    // Terminador
    const totalDataBits = getTotalDataBits(version);
    const terminatorLen = Math.min(4, totalDataBits - dataBits.length);
    for (let i = 0; i < terminatorLen; i++) dataBits.push(0);
    
    // Rellenar hasta múltiplo de 8
    while (dataBits.length % 8 !== 0) dataBits.push(0);
    
    // Convertir a bytes
    for (let i = 0; i < dataBits.length; i += 8) {
      let byte = 0;
      for (let j = 0; j < 8; j++) {
        byte = (byte << 1) | (dataBits[i + j] || 0);
      }
      bytes.push(byte);
    }
    
    // Rellenar con patrones
    const totalBytes = Math.floor(totalDataBits / 8);
    let pad = 0;
    const padBytes = [0xEC, 0x11];
    while (bytes.length < totalBytes) {
      bytes.push(padBytes[pad % 2]);
      pad++;
    }
    
    return bytes;
  }

  function getTotalDataBits(version) {
    const totalCodewords = getTotalCodewords(version);
    const ecCodewords = EC_CODEWORDS_PER_BLOCK[version]?.L || 7;
    return (totalCodewords - ecCodewords) * 8;
  }

  function getTotalCodewords(version) {
    const size = version * 4 + 17;
    let total = size * size;
    // Restar módulos de función (simplificado)
    total -= 8 * 8 * 3; // Finders
    total -= 15 * 2 + 9; // Timing
    total -= 31; // Format info
    if (version >= 2) total -= 25; // Alignment
    return Math.floor(total / 8);
  }

  // Generar matriz QR simplificada (solo para demostración visual)
  function generateMatrix(text) {
    const version = getVersion(text.length);
    if (version === -1) return null;

    const size = version * 4 + 17;
    const matrix = Array(size).fill(null).map(() => Array(size).fill(null));
    const reserved = Array(size).fill(null).map(() => Array(size).fill(false));

    // Dibujar patrones de finder (esquinas)
    function drawFinder(row, col) {
      for (let r = -1; r <= 7; r++) {
        for (let c = -1; c <= 7; c++) {
          const rr = row + r, cc = col + c;
          if (rr < 0 || rr >= size || cc < 0 || cc >= size) continue;
          
          let val = 0;
          if (r >= 0 && r <= 6 && c >= 0 && c <= 6) {
            if (r === 0 || r === 6 || c === 0 || c === 6) val = 1;
            else if (r >= 2 && r <= 4 && c >= 2 && c <= 4) val = 1;
            else val = 0;
          }
          matrix[rr][cc] = val;
          reserved[rr][cc] = true;
        }
      }
    }

    drawFinder(0, 0);
    drawFinder(0, size - 7);
    drawFinder(size - 7, 0);

    // Patrones de sincronización (timing)
    for (let i = 8; i < size - 8; i++) {
      matrix[6][i] = matrix[i][6] = i % 2 === 0 ? 1 : 0;
      reserved[6][i] = reserved[i][6] = true;
    }

    // Información de formato (simplificada - nivel L, máscara 0)
    // Estos son los bits precalculados para formato L-0
    const formatBits = [1,1,1,0,1,1,1,1,1,0,0,0,1,0,0];
    for (let i = 0; i < 15; i++) {
      // Posiciones alrededor del finder superior izquierdo
      if (i < 6) matrix[i][8] = formatBits[i];
      else if (i < 8) matrix[i + 1][8] = formatBits[i];
      else matrix[8][14 - i] = formatBits[i];
      
      // Posiciones alrededor de los otros finders
      if (i < 8) matrix[8][size - 1 - i] = formatBits[i];
      else matrix[size - 7 + (i - 8)][8] = formatBits[i];
      
      reserved[8][i] = true;
      if (i < 8) reserved[8][size - 1 - i] = true;
      else reserved[size - 7 + (i - 8)][8] = true;
    }
    matrix[size - 8][8] = 1; // Módulo oscuro
    reserved[size - 8][8] = true;

    // Codificar y colocar datos (simplificado)
    const dataBytes = encodeData(text, version);
    let bitIndex = 0;
    const allBits = [];
    for (const byte of dataBytes) {
      for (let i = 7; i >= 0; i--) {
        allBits.push((byte >> i) & 1);
      }
    }

    // Colocar datos en zigzag (simplificado)
    let col = size - 1;
    let direction = -1; // -1 = arriba, 1 = abajo
    let row = size - 1;

    while (col > 0) {
      if (col === 6) col--; // Saltar columna de timing
      
      for (let i = 0; i < 2; i++) {
        const c = col - i;
        if (c < 0 || reserved[row][c]) continue;
        
        if (bitIndex < allBits.length) {
          matrix[row][c] = allBits[bitIndex++];
        } else {
          matrix[row][c] = 0;
        }
      }
      
      row += direction;
      if (row < 0 || row >= size) {
        direction = -direction;
        row += direction;
        col -= 2;
      }
    }

    // Aplicar máscara 0 (patrón simple)
    for (let r = 0; r < size; r++) {
      for (let c = 0; c < size; c++) {
        if (!reserved[r][c] && (r + c) % 2 === 0) {
          matrix[r][c] ^= 1;
        }
      }
    }

    return matrix;
  }

  // Función pública: generar QR como imagen
  function generate(text, tamano = 200, colorModulo = '#000000', colorFondo = '#ffffff') {
    const matrix = generateMatrix(text);
    if (!matrix) return null;

    const size = matrix.length;
    const canvas = document.createElement('canvas');
    const padding = 4; // Módulos de silencio
    const totalSize = size + padding * 2;
    const moduloSize = tamano / totalSize;

    canvas.width = canvas.height = tamano;
    const ctx = canvas.getContext('2d');

    // Fondo
    ctx.fillStyle = colorFondo;
    ctx.fillRect(0, 0, tamano, tamano);

    // Módulos
    ctx.fillStyle = colorModulo;
    for (let r = 0; r < size; r++) {
      for (let c = 0; c < size; c++) {
        if (matrix[r][c] === 1) {
          ctx.fillRect(
            (c + padding) * moduloSize,
            (r + padding) * moduloSize,
            moduloSize,
            moduloSize
          );
        }
      }
    }

    return canvas.toDataURL();
  }

  return { generate: generate };
})();

// 🔐 FUNCIÓN HASH SHA-256 NATIVO (CORREGIDA)
// ============================================
async function hashLocal(data, longitudCompleta = false) {
  if (!data) return 'none';
  
  if (typeof crypto === 'undefined' || !crypto.subtle) {
    log('⚠️ crypto.subtle no disponible (requiere HTTPS)', 'warning');
    return 'none';
  }

  try {
    const encoder = new TextEncoder();
    const buffer = encoder.encode(String(data));
    const hashBuffer = await crypto.subtle.digest('SHA-256', buffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    
    return longitudCompleta ? hashHex : hashHex.slice(0, 16);
  } catch (error) {
    log('⚠️ Error generando hash: ' + error.message, 'warning');
    return 'none';
  }
}

// Función para dibujar gráfico de líneas nativo
function drawLineChart(canvasId, labels, data, color = '#00e676') {
  const canvas = document.getElementById(canvasId);
  if (!canvas) return null;

  const ctx = canvas.getContext('2d');
  const width = canvas.width;
  const height = canvas.height;
  const padding = 40;

  // Limpiar canvas
  ctx.clearRect(0, 0, width, height);

  if (!data || data.length === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('Sin datos disponibles', width / 2, height / 2);
    return null;
  }

  // Calcular escalas
  const maxValue = Math.max(...data);
  const minValue = Math.min(0, ...data);
  const range = maxValue - minValue || 1;

  // Dibujar ejes
  ctx.strokeStyle = 'rgba(255,255,255,0.1)';
  ctx.lineWidth = 1;
  
  // Eje Y
  ctx.beginPath();
  ctx.moveTo(padding, padding);
  ctx.lineTo(padding, height - padding);
  ctx.stroke();
  
  // Eje X
  ctx.beginPath();
  ctx.moveTo(padding, height - padding);
  ctx.lineTo(width - padding, height - padding);
  ctx.stroke();

  // Dibujar líneas de grid horizontales
  ctx.strokeStyle = 'rgba(255,255,255,0.05)';
  for (let i = 0; i <= 4; i++) {
    const y = padding + (height - 2 * padding) * (i / 4);
    ctx.beginPath();
    ctx.moveTo(padding, y);
    ctx.lineTo(width - padding, y);
    ctx.stroke();
    
    // Etiquetas Y
    const valor = maxValue - (range * i / 4);
    ctx.fillStyle = '#8b95a5';
    ctx.font = '10px system-ui';
    ctx.textAlign = 'right';
    ctx.fillText('$' + valor.toFixed(0), padding - 5, y + 3);
  }

  // Dibujar línea de datos
  const stepX = (width - 2 * padding) / (data.length - 1 || 1);
  
  // Área bajo la línea
  ctx.fillStyle = color + '20'; // 20 = 12% opacidad
  ctx.beginPath();
  ctx.moveTo(padding, height - padding);
  
  data.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    ctx.lineTo(x, y);
  });
  
  ctx.lineTo(padding + (data.length - 1) * stepX, height - padding);
  ctx.closePath();
  ctx.fill();

  // Línea principal
  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.beginPath();
  
  data.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    
    if (i === 0) {
      ctx.moveTo(x, y);
    } else {
      ctx.lineTo(x, y);
    }
  });
  
  ctx.stroke();

  // Puntos en cada dato
  ctx.fillStyle = color;
  data.forEach((valor, i) => {
    const x = padding + i * stepX;
    const y = height - padding - ((valor - minValue) / range) * (height - 2 * padding);
    
    ctx.beginPath();
    ctx.arc(x, y, 3, 0, Math.PI * 2);
    ctx.fill();
  });

  // Etiquetas X (solo algunas para no saturar)
  ctx.fillStyle = '#8b95a5';
  ctx.font = '10px system-ui';
  ctx.textAlign = 'center';
  
  const step = Math.max(1, Math.floor(labels.length / 6));
  labels.forEach((label, i) => {
    if (i % step === 0 || i === labels.length - 1) {
      const x = padding + i * stepX;
      ctx.fillText(label.substring(5), x, height - padding + 15); // Mostrar solo DD-MM
    }
  });

  return { canvas, ctx };
}

// Función para dibujar gráfico de dona nativo
function drawDoughnutChart(canvasId, labels, data, colors = ['#00e676', '#29b6f6', '#ab47bc', '#ffa726', '#66bb6a']) {
  const canvas = document.getElementById(canvasId);
  if (!canvas) return null;

  const ctx = canvas.getContext('2d');
  const width = canvas.width;
  const height = canvas.height;
  const centerX = width / 2;
  const centerY = height / 2;
  const radius = Math.min(width, height) / 2 - 40;
  const innerRadius = radius * 0.6;

  // Limpiar canvas
  ctx.clearRect(0, 0, width, height);

  if (!data || data.length === 0) {
    ctx.fillStyle = '#8b95a5';
    ctx.font = '14px system-ui';
    ctx.textAlign = 'center';
    ctx.fillText('Sin datos disponibles', centerX, centerY);
    return null;
  }

  const total = data.reduce((sum, val) => sum + val, 0);
  let startAngle = -Math.PI / 2; // Empezar desde arriba

  // Dibujar segmentos
  data.forEach((valor, i) => {
    const sliceAngle = (valor / total) * 2 * Math.PI;
    const endAngle = startAngle + sliceAngle;

    ctx.fillStyle = colors[i % colors.length];
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius, startAngle, endAngle);
    ctx.arc(centerX, centerY, innerRadius, endAngle, startAngle, true);
    ctx.closePath();
    ctx.fill();

    startAngle = endAngle;
  });

  // Texto central
  ctx.fillStyle = '#ffffff';
  ctx.font = 'bold 20px system-ui';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(total.toString(), centerX, centerY - 10);
  
  ctx.fillStyle = '#8b95a5';
  ctx.font = '12px system-ui';
  ctx.fillText('Total', centerX, centerY + 10);

  // Leyenda
  const legendY = height - 20;
  const legendSpacing = width / (labels.length + 1);
  
  ctx.font = '10px system-ui';
  labels.forEach((label, i) => {
    const x = legendSpacing * (i + 1);
    
    // Cuadro de color
    ctx.fillStyle = colors[i % colors.length];
    ctx.fillRect(x - 30, legendY - 5, 10, 10);
    
    // Texto
    ctx.fillStyle = '#e0e0e0';
    ctx.textAlign = 'left';
    ctx.fillText(label.substring(0, 10), x - 15, legendY + 3);
  });

  return { canvas, ctx };
}

async function exportFinancialData(formato = 'csv') {
  try {
    showNotification('🔄 Preparando exportación...', 'info');
    
    const respuesta = await req('/api/analytics/financial?period=30d');
    
    if (!respuesta || !respuesta.success || !respuesta.data) {
      showNotification('❌ No hay datos para exportar', 'error');
      return;
    }

    const datos = respuesta.data;
    let contenido = '';
    let nombreArchivo = '';
    let tipoMIME = '';

    if (formato === 'csv') {
      // Exportar como CSV
      nombreArchivo = `analytics_${new Date().toISOString().split('T')[0]}.csv`;
      tipoMIME = 'text/csv';
      
      contenido = 'Fecha,Ingresos,Ventas\n';
      if (Array.isArray(datos.revenueByDay)) {
        datos.revenueByDay.forEach(d => {
          contenido += `${d.date},${d.amount},${d.sales || 0}\n`;
        });
      }
      
      contenido += '\n\nCategoría,Cantidad\n';
      if (Array.isArray(datos.salesByCategory)) {
        datos.salesByCategory.forEach(c => {
          contenido += `${c.category},${c.count}\n`;
        });
      }
      
    } else if (formato === 'json') {
      // Exportar como JSON
      nombreArchivo = `analytics_${new Date().toISOString().split('T')[0]}.json`;
      tipoMIME = 'application/json';
      contenido = JSON.stringify(datos, null, 2);
      
    } else {
      showNotification('❌ Formato no soportado', 'error');
      return;
    }

    // Crear blob y descargar
    const blob = new Blob([contenido], { type: tipoMIME });
    const url = URL.createObjectURL(blob);
    const enlace = document.createElement('a');
    enlace.href = url;
    enlace.download = nombreArchivo;
    document.body.appendChild(enlace);
    enlace.click();
    document.body.removeChild(enlace);
    URL.revokeObjectURL(url);

    showNotification(`✅ Archivo ${formato.toUpperCase()} descargado`, 'success');
    
  } catch (error) {
    log('❌ Error exportando datos: ' + error.message, 'error');
    showNotification('❌ Error: ' + error.message, 'error');
  }
}

async function loadFeed(filter = 'all', reset = true) {
  if (reset) feedPage = 1;
  
  const container = document.getElementById('feedContainer');
  if (!container) return;
  
  if (reset) container.innerHTML = '
Cargando...
';
  
  try {
    const res = await fetch(`/api/feed?page=${feedPage}&limit=10&filter=${filter}&followedOnly=${followedOnly}`, {
      headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` }
    });
    const data = await res.json();
    
    if (!data.success) throw new Error(data.error);
    
    if (reset && data.posts.length === 0) {
      container.innerHTML = '
📭 Aún no hay productos en esta categoría. ¡Sé el primero en publicar!
';
      return;
    }
    
    // Renderizar posts
    data.posts.forEach(post => {
      const postEl = document.createElement('div');
      postEl.className = 'glass';
      postEl.style.padding = '16px';
      postEl.style.borderRadius = '12px';
      postEl.innerHTML = `

${new Date(post.createdAt).toLocaleDateString()}

          

        

        
        
${post.titulo}

        
${post.descripcion}

        
        ${post.multimedia.fileName ? `
          

            

              📎 ${post.multimedia.fileName} • ${(post.multimedia.fileSize / 1024 / 1024).toFixed(2)} MB
            

          

        ` : ''}
        
        

          $${post.precio} USD
          📦 ${post.categoria}
        

        
        

          
          
          
        

        
        
      `;

    });

`;
  }
}

async function publishProduct(e) {

  formData.append('price', document.getElementById('pubPrecio').value);
  
  const file = document.getElementById('pubArchivo').files[0];
  if (file) formData.append('archivo', file);
  
  try {
    const res = await fetch('/vault', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` },
      body: formData
    });

      alert('✅ ¡Producto publicado exitosamente!');
      closePublishModal();
      loadFeed('all', true); // Recargar feed
    } else {
      alert('❌ Error: ' + data.error);
    }
  } catch (err) {
    alert('❌ Error de conexión: ' + err.message);
  }
}
